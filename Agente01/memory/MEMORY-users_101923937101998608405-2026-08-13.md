```sql
DECLARE @var_hoy DATETIME

SET @var_hoy = GETDATE()

SELECT
    AC.prepaga,
    AC.contra,
    AC.vigen_desde,
    AC.cuenta,
    AC.subcta,
    mega_cuenta_anterior = AC.mega_cuenta,
    mega_cuenta_nueva = C.mega_cuenta

INTO #contratos_afectados

FROM AFI_CLASE AS AC

    INNER JOIN CUENTAS AS C
    ON C.cuenta = AC.cuenta

WHERE AC.cuenta = 1198
AND AC.subcta = '11982'
AND AC.vigen_desde = (
                        SELECT MAX(vigen_desde)
                        FROM AFI_CLASE
                        WHERE prepaga = AC.prepaga
                        AND contra = AC.contra
                        AND vigen_desde <= GETDATE()
                      )
AND (AC.baja_fecha IS NULL OR AC.baja_fecha > GETDATE())
AND AC.mega_cuenta <> C.mega_cuenta


INSERT INTO PREFACT_FILTRO_MEGA_CUENTA
    (
    tipo,
    identificador,
    mega_cuenta,
    alta_fecha,
    modi_fecha,
    baja_fecha
    )
SELECT DISTINCT
    tipo = 'I',
    identificador = CUE.identificador,
    mega_cuenta = CA.mega_cuenta_nueva,
    alta_fecha = @var_hoy,
    modi_fecha = @var_hoy,
    baja_fecha = NULL

FROM PREFACT_PLANTILLA AS P

    INNER JOIN PREFACT_FILTRO_CUENTA AS CUE
    ON CUE.tipo = 'I'
    AND CUE.identificador = P.cod_plantilla
    AND CUE.cuenta = 1198
    AND CUE.baja_fecha IS NULL

    INNER JOIN PREFACT_FILTRO_SUBCUENTA AS SUB
    ON SUB.tipo = CUE.tipo
    AND SUB.identificador = CUE.identificador
    AND SUB.cuenta = 1198
    AND SUB.subcta = '11982'
    AND SUB.baja_fecha IS NULL

    INNER JOIN #contratos_afectados AS CA
    ON CA.cuenta = CUE.cuenta

WHERE P.baja_fecha IS NULL
AND NOT EXISTS (
                SELECT 1
                FROM PREFACT_FILTRO_MEGA_CUENTA
                WHERE tipo = CUE.tipo
                AND identificador = CUE.identificador
                AND mega_cuenta = CA.mega_cuenta_nueva
                AND baja_fecha IS NULL
               )


UPDATE AC
SET
    mega_cuenta = CA.mega_cuenta_nueva,
    modi_fecha = @var_hoy

FROM AFI_CLASE AS AC

    INNER JOIN #contratos_afectados AS CA
    ON CA.prepaga = AC.prepaga
    AND CA.contra = AC.contra
    AND CA.vigen_desde = AC.vigen_desde

WHERE AC.mega_cuenta = CA.mega_cuenta_anterior


SELECT *
FROM #contratos_afectados

DROP TABLE #contratos_afectados
```
