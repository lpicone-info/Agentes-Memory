4. Pegá esto y ejecutalo con `F5`:

```vb
Sub NegritaAntesDelIgual()
    Dim p As Paragraph
    Dim r As Range
    Dim pos As Long

    For Each p In ActiveDocument.Paragraphs
        pos = InStr(p.Range.Text, "=")

        If pos > 1 Then
            Set r = p.Range.Duplicate
            r.End = r.Start + pos - 2
            r.Font.Bold = True
        End If
    Next p
End Sub
```
