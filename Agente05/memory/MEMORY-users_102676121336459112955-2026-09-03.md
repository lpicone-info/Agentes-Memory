afiliados_con_mas_de_un_plan AS
(
    SELECT
        prepaga,
        contra,
        inte
    FROM credenciales_vigentes
    GROUP BY
        prepaga,
        contra,
        inte
    HAVING COUNT(*) > 1
       AND COUNT(DISTINCT plan_codi) > 1
)
SELECT
    cv.prepaga,
    cv.contra,
    cv.inte,
    cv.creden,
    cv.plan_codi,
    cv.vigen_desde,
    cv.vigen_hasta,
    cv.baja_fecha,
    cv.nro_emision,
    cv.creden_uso,
    cv.ult_creden_estado,
    cv.alta_fecha,
    cv.modi_fecha
FROM credenciales_vigentes cv
INNER JOIN afiliados_con_mas_de_un_plan af
        ON af.prepaga = cv.prepaga
       AND af.contra = cv.contra

ORDER BY
