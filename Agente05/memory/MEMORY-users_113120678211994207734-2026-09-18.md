```sql
IF OBJECT_ID('dbo.prs_solicitud_auto_finalizada_por_autorizacion') IS NOT NULL

/*
  PROCEDURE       : prs_solicitud_auto_finalizada_por_autorizacion
  Creation DATE   : 20260918
  Programmer      : whiguera
  Description     : Obtiene la solicitud activa asociada a una autorización
                    y determina si se encuentra en estado Finalizada.
  Reference       : Tarea #29489

  Consideraciones:
    1. Solo considera autorizaciones vigentes.
    2. Solo considera relaciones vigentes.
    3. Solo considera solicitudes vigentes.
    4. No modifica datos.
    5. No genera adjuntos.
    6. No ejecuta notificaciones.
    7. Devuelve siempre una fila de resultado.
*/

CREATE PROCEDURE dbo.prs_solicitud_auto_finalizada_por_autorizacion
(
	@prm_sucur	SMALLINT,
	@prm_auto	INTEGER
)
AS
BEGIN

	SET NOCOUNT ON

	DECLARE @var_cod_error				INTEGER,
			@var_deno_error				VARCHAR(200),
			@var_soli_auto				NUMERIC(18,0),
			@var_soli_esta				CHAR(1),
			@var_es_finalizada			CHAR(1),
			@var_cantidad_solicitudes	INTEGER

	SET @var_cod_error = 0
	SET @var_deno_error = ''
	SET @var_soli_auto = NULL
	SET @var_soli_esta = NULL
	SET @var_es_finalizada = 'N'
	SET @var_cantidad_solicitudes = 0

	/*
	  Validación de parámetros.
	*/
	IF @prm_sucur IS NULL
	OR @prm_auto IS NULL
	BEGIN
		SET @var_cod_error = -1
		SET @var_deno_error =
			'La sucursal y la autorización son obligatorias'

		GOTO FIN
	END

	/*
	  La autorización debe existir y encontrarse vigente.
	*/
	IF NOT EXISTS
	(
		SELECT 1
		FROM AUTORIZACIONES A
		WHERE A.sucur = @prm_sucur
		AND A.auto = @prm_auto
		AND A.baja_fecha IS NULL
	)
	BEGIN
		SET @var_cod_error = -2
		SET @var_deno_error =
			'La autorización no existe o se encuentra dada de baja'

		GOTO FIN
	END

	/*
	  Se cuentan las solicitudes activas relacionadas con la autorización.

	  La validación evita elegir arbitrariamente una solicitud cuando
	  existan varias relaciones activas para el mismo sucur + auto.
	*/
	SELECT
		@var_cantidad_solicitudes =
			COUNT(DISTINCT SAA.soli_auto)
	FROM SOLICITUDES_AUTO_AUTORIZACIONES SAA

	INNER JOIN SOLICITUDES_AUTO SA
		ON SA.soli_auto = SAA.soli_auto
		AND SA.baja_fecha IS NULL

	WHERE SAA.sucur = @prm_sucur
	AND SAA.auto = @prm_auto
	AND SAA.baja_fecha IS NULL

	/*
	  Una autorización sin solicitud asociada no representa un error
	  técnico. En ese caso no corresponde ejecutar la notificación.
	*/
	IF @var_cantidad_solicitudes = 0
	BEGIN
		SET @var_deno_error =
			'La autorización no posee una solicitud activa asociada'

		GOTO FIN
	END

	/*
	  La autorización no debe pertenecer simultáneamente a varias
	  solicitudes activas.
	*/
	IF @var_cantidad_solicitudes > 1
	BEGIN
		SET @var_cod_error = -3
		SET @var_deno_error =
			'La autorización posee más de una solicitud activa asociada'

		GOTO FIN
	END

	/*
	  Como ya se confirmó que existe una única solicitud activa,
	  MAX permite obtenerla sin recurrir a un TOP 1 arbitrario.
	*/
	SELECT
		@var_soli_auto = MAX(SAA.soli_auto)
	FROM SOLICITUDES_AUTO_AUTORIZACIONES SAA

	INNER JOIN SOLICITUDES_AUTO SA
		ON SA.soli_auto = SAA.soli_auto
		AND SA.baja_fecha IS NULL

	WHERE SAA.sucur = @prm_sucur
	AND SAA.auto = @prm_auto
	AND SAA.baja_fecha IS NULL

	SELECT
		@var_soli_esta = SA.soli_esta
	FROM SOLICITUDES_AUTO SA
	WHERE SA.soli_auto = @var_soli_auto
	AND SA.baja_fecha IS NULL

	IF @var_soli_esta = 'F'
	BEGIN
		SET @var_es_finalizada = 'S'
		SET @var_deno_error =
			'La solicitud asociada se encuentra finalizada'
	END
	ELSE
	BEGIN
		SET @var_es_finalizada = 'N'
		SET @var_deno_error =
			'La solicitud asociada no se encuentra finalizada'
	END

	FIN:

	SELECT
		cod_error		= @var_cod_error,
		deno_error		= @var_deno_error,
		soli_auto		= @var_soli_auto,
		soli_esta		= @var_soli_esta,
		es_finalizada	= @var_es_finalizada

END
GO
```

# 9. No agregar transacciones en este bloque

No agregues:

```powerscript
COMMIT;
```

ni:

```powerscript
ROLLBACK;
```

El evento ya realizó el `COMMIT` de la autorización antes de llegar a este punto:

```powerscript
COMMIT;
SQLCA.AUTOCOMMIT = TRUE
```

Por lo tanto:

```text
Error de notificación ≠ error de grabación de autorización
```

Si la notificación falla:

1. La autorización permanece grabada.
2. La solicitud permanece en `F`.
3. El usuario recibe el mensaje de error.
4. Puede reintentar desde `wp_soli_reg → Enviar Notificación`.
