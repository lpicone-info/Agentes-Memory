/*
  PROCEDURE       : pru_solicitudes_auto_notif_finalizada
  Creation DATE   : 20260915
  Programmer      : whiguera
  Description     : Coordina las notificaciones de una solicitud finalizada
                    luego de que PowerBuilder haya confirmado la transacción
                    de finalización y haya preparado los adjuntos PDF.
  Reference       : Tarea #29489

  Responsabilidades:
    1. Validar que la solicitud esté en estado F.
    2. Enviar el proceso 5 al prestador, sin adjuntos, cuando el origen
       funcional de la solicitud sea prestador.
    3. Determinar la notificación al afiliado:
         - Proceso 8  : todas las autorizaciones activas autorizadas.
         - Proceso 9  : todas las autorizaciones activas negadas.
         - Proceso 13 : solicitud mixta.
         - Proceso 23 : solicitud finalizada ingresada por prestador GIS.
    4. Enviar el proceso del afiliado pasando el XML generado por el NVO.
    5. Evitar duplicar notificaciones ya auditadas.

  Contrato del XML:

    <d_soli_auto_adjuntos_mail_xml>

        <tipo_adjunto>...</tipo_adjunto>
        <file_name>...</file_name>
        <archivo_path_full>...</archivo_path_full>
      </d_soli_auto_adjuntos_mail_xml_row>
    </d_soli_auto_adjuntos_mail_xml>

  Importante:
    - El proceso 5 se invoca sin @prm_adjuntos.
    - Los procesos 8, 9, 13 y 23 reciben @prm_adjuntos.
    - Los procesos 27 y 28 no se ejecutan aquí porque corresponden a
      notificaciones parciales de solicitudes en estado V.
    - Este procedimiento no modifica SOLICITUDES_AUTO.
    - Este procedimiento no ejecuta COMMIT ni ROLLBACK.
    - La transacción de finalización debe haber sido confirmada antes
      del llamado a este procedimiento.
    - Se respetan las convenciones SQL definidas para la tarea.

CREATE PROCEDURE dbo.pru_solicitudes_auto_notif_finalizada
(
	@prm_soli_auto			NUMERIC(18,0),
	@prm_adjuntos				XML,
	@prm_usuari					CHAR(15) 			= NULL,
	@prm_ya_notificada	CHAR(1) 			= NULL OUTPUT,
	@prm_mensaje				VARCHAR(500) 	= NULL OUTPUT
)
AS
BEGIN

	SET NOCOUNT ON
	SET ANSI_NULLS ON
	SET ANSI_WARNINGS ON
	SET ANSI_PADDING ON
	SET CONCAT_NULL_YIELDS_NULL ON
	SET QUOTED_IDENTIFIER ON
	SET ARITHABORT ON
	SET NUMERIC_ROUNDABORT OFF
	
	

	DECLARE @var_usuari										CHAR(15),
					@var_soli_esta								CHAR(1),
					@var_tipo_usuario							CHAR(1),
					@var_origen										CHAR(1),
					@var_cantidad_autorizaciones	INT,
					@var_cantidad_aprobadas				INT,
					@var_cantidad_negadas					INT,
					@var_proceso_afiliado					SMALLINT

	SET @prm_ya_notificada 	= 'N'
	SET @prm_mensaje 				= ''
	
	SET @var_usuari 				= ISNULL(@prm_usuari, SUSER_SNAME())

	BEGIN TRY

		/*
		  Se lee el estado, el tipo de usuario que originó la solicitud y
		  el origen técnico registrado en SOLICITUDES_AUTO.

		  En el modelo vigente:
		    - origen = 'G' identifica una solicitud ingresada por GIS.
		    - tipo_usuario = 'P' identifica el origen funcional prestador.
		*/
		SELECT	@var_soli_esta		= SA.soli_esta,
						@var_tipo_usuario	= SA.tipo_usuario,
						@var_origen				= SA.origen
		FROM SOLICITUDES_AUTO SA
		WHERE SA.soli_auto = @prm_soli_auto
		AND SA.baja_fecha IS NULL

		IF @var_soli_esta IS NULL
		BEGIN
			RAISERROR(
				'La solicitud indicada no existe o se encuentra dada de baja',
				16,
				1
			)
			RETURN
		END

		IF @var_soli_esta <> 'F'
		BEGIN
			RAISERROR(
				'La solicitud no se encuentra en estado FINALIZADA',
				16,
				1
			)
			RETURN
		END

		/*
		  El XML debe corresponder a la misma solicitud que se está
		  notificando. La validación se ejecuta solo cuando se recibe XML.
		*/
		IF @prm_adjuntos IS NOT NULL
		AND EXISTS (
			SELECT 1
			FROM @prm_adjuntos.nodes(
				'/d_soli_auto_adjuntos_mail_xml/d_soli_auto_adjuntos_mail_xml_row'
			) AS T(c)
			WHERE T.c.value('(soli_auto)[1]', 'NUMERIC(18,0)') <> @prm_soli_auto
		)
		BEGIN
			RAISERROR(
				'El XML de adjuntos no corresponde a la solicitud finalizada',
				16,
				1
			)
			RETURN
		END

		/*
		  Se cuentan únicamente las autorizaciones vigentes:
		    - relación vigente en SOLICITUDES_AUTO_AUTORIZACIONES.
		    - autorización vigente en AUTORIZACIONES.

		  Los estados utilizados para clasificar la notificación son:
		    A = autorizada
		    N = negada
		*/
		SELECT	@var_cantidad_autorizaciones = COUNT(*),
						@var_cantidad_aprobadas =
							ISNULL(SUM(CASE WHEN A.esta = 'A' THEN 1 ELSE 0 END), 0),
						@var_cantidad_negadas =
							ISNULL(SUM(CASE WHEN A.esta = 'N' THEN 1 ELSE 0 END), 0)
		FROM SOLICITUDES_AUTO_AUTORIZACIONES SAA
		INNER JOIN AUTORIZACIONES A
			ON A.sucur = SAA.sucur
			AND A.auto = SAA.auto
			AND A.baja_fecha IS NULL
		WHERE SAA.soli_auto = @prm_soli_auto
		AND SAA.baja_fecha IS NULL

		IF @var_cantidad_autorizaciones = 0
		BEGIN
			RAISERROR(
				'La solicitud finalizada no tiene autorizaciones vigentes para notificar',
				16,
				1
			)
			RETURN
		END

		/*
		  No se debe clasificar automáticamente una solicitud que conserve
		  autorizaciones en estados diferentes de A o N.
		*/
		IF EXISTS (
			SELECT 1
			FROM SOLICITUDES_AUTO_AUTORIZACIONES SAA
			INNER JOIN AUTORIZACIONES A
				ON A.sucur = SAA.sucur
				AND A.auto = SAA.auto
				AND A.baja_fecha IS NULL
			WHERE SAA.soli_auto = @prm_soli_auto
			AND SAA.baja_fecha IS NULL
			AND ISNULL(A.esta, '') NOT IN ('A', 'N')
		)
		BEGIN
			RAISERROR(
				'La solicitud contiene autorizaciones en estados no notificables',
				16,
				1
			)
			RETURN
		END

		/*
		  Si el origen es prestador, se notifica el proceso 5 al prestador.
		  El proceso 5 no recibe XML y por lo tanto no lleva PDF.
		*/
		IF @var_tipo_usuario = 'P'
		AND NOT EXISTS (
			SELECT 1
			FROM SOLICITUDES_AUTO_AUDITORIA
			WHERE soli_auto = @prm_soli_auto
			AND activi LIKE '%proceso_id: 5%'
		)
		BEGIN
			SET @prm_ya_notificada = 'S'
			EXEC dbo.pri_solicitudes_auto_notif
				@prm_id        = 5,
				@prm_soli_auto = @prm_soli_auto,
				@prm_usuari    = @var_usuari
		END
		ELSE
		BEGIN
			SET @prm_ya_notificada = 'S'
			SET @prm_mensaje =
				'No corresponde enviar nuevamente la notificacion de esta solicitud.'
		END

		/*
		  Proceso 23:
		    Corresponde a una solicitud finalizada ingresada por prestador
		    GIS. En el modelo actual esta condición se identifica por:
		      origen = 'G'
		      tipo_usuario = 'P'

		  La condición tiene prioridad sobre 8, 9 y 13 porque el proceso 23
		  es la notificación específica para ese origen.
		*/
		IF @var_origen = 'G'
		AND @var_tipo_usuario = 'P'
		BEGIN
			SET @var_proceso_afiliado = 23
		END
		ELSE IF @var_cantidad_aprobadas = @var_cantidad_autorizaciones
		BEGIN
			SET @var_proceso_afiliado = 8
		END
		ELSE IF @var_cantidad_negadas = @var_cantidad_autorizaciones
		BEGIN
			SET @var_proceso_afiliado = 9
		END
		ELSE IF @var_cantidad_aprobadas > 0
		AND @var_cantidad_negadas > 0
		BEGIN
			SET @var_proceso_afiliado = 13
		END

		IF @var_proceso_afiliado IS NULL
		BEGIN
			RAISERROR(
				'No fue posible determinar el proceso de notificación al afiliado',
				16,
				1
			)
			RETURN
		END

		/*
		  Se envía una sola notificación al afiliado para el proceso
		  determinado. El XML se entrega a pri_solicitudes_auto_notif,
		  que registra cada file_name y archivo_path_full mediante
		  afi_mail_attach_binary antes de enviar el correo.
		*/
		IF NOT EXISTS (
			SELECT 1
			FROM SOLICITUDES_AUTO_AUDITORIA
			WHERE soli_auto = @prm_soli_auto
			AND activi LIKE '%proceso_id: '
				+ CONVERT(VARCHAR(10), @var_proceso_afiliado)
				+ '%'
		)
		BEGIN
			SET @prm_ya_notificada = 'N'
			
			EXEC dbo.pri_solicitudes_auto_notif
				@prm_id        = @var_proceso_afiliado,
				@prm_soli_auto = @prm_soli_auto,
				@prm_usuari    = @var_usuari,
				@prm_adjuntos  = @prm_adjuntos
		END
		ELSE
		BEGIN
			SET @prm_ya_notificada = 'S'
			SET @prm_mensaje =
				'No corresponde enviar nuevamente la notificacion de esta solicitud.'
		END

	END TRY
	BEGIN CATCH
    DECLARE @var_error_mensaje   NVARCHAR(4000),
            @var_error_severidad INT,
            @var_error_estado    INT

    SELECT  @var_error_mensaje   = ERROR_MESSAGE(),
            @var_error_severidad = ERROR_SEVERITY(),
            @var_error_estado    = ERROR_STATE()

    RAISERROR(@var_error_mensaje, @var_error_severidad, @var_error_estado)
	END CATCH
END
GO
