SET ANSI_NULLS ON
GO

GO

CREATE PROCEDURE dbo.prs_tmp_cober_copa_grupos_rango_valor_export_8
(
    @prm_prepaga       SMALLINT,
    @prm_vigen_desde   DATETIME,
    @prm_planes        XML
)
AS
BEGIN

    SET NOCOUNT ON

    SET CONCAT_NULL_YIELDS_NULL ON
    SET ANSI_WARNINGS ON
    SET ANSI_NULLS ON
    SET QUOTED_IDENTIFIER ON

    SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

    CREATE TABLE #PLANES
    (
        plan_codi CHAR(10) NOT NULL
    )

    INSERT INTO #PLANES
    (
        plan_codi
    )
    SELECT
        plan_codi = N.value('.', 'CHAR(10)')
    FROM @prm_planes.nodes('x') AS T(N)

    SELECT
        id_rango_valores = CCG.cod,
        rango_id         = CCGRV.rango_id,
        valor            = CONVERT(NUMERIC(12), CCGRV.valor)
    FROM COBER_COPA_GRUPOS CCG

        INNER JOIN PLANES P
        ON P.plan_codi = CCG.plan_codi

        INNER JOIN PLANES_GRUPOS PG
        ON PG.plan_grupo = P.plan_grupo

        INNER JOIN #PLANES T
        ON T.plan_codi = CCG.plan_codi

        INNER JOIN COBER_COPA_GRUPOS_RANGO_VALOR CCGRV
        ON CCGRV.cod_grupo = CCG.cod

        INNER JOIN RANGO_GENERICO RG
        ON RG.id = CCGRV.rango_id

    WHERE CCG.baja_fecha IS NULL
    AND P.prepaga = @prm_prepaga
    AND CCG.vigen_desde >= @prm_vigen_desde
    AND CCG.tipo_grupo = 'S'
    AND CCGRV.baja_fecha IS NULL
    AND CCG.prestad IS NULL

    ORDER BY
        CCG.vigen_desde,
        CCG.cod,
        CCGRV.rango_id

    DROP TABLE #PLANES

END
GO

# Por qué no conviene reutilizar las variables de `ue_agregar`

Aunque tengan el mismo nombre, esto no funciona:

```powerscript
// ue_agregar
li_prepaga = ...
```

y después:

```powerscript
// ue_aceptar
lds_rango_export.Retrieve(li_prepaga, ...)
```

Cada evento tiene su propio ámbito. Al finalizar `ue_agregar`, sus variables locales dejan de existir.

La solución correcta es:

```text
ue_agregar:
    valida y carga datos para recuperar la pantalla

ue_aceptar:
    vuelve a leer los filtros actuales
    recupera el DataStore
    exporta el CSV
```

Así también se garantiza que, si el usuario cambió la prepaga, la vigencia o los planes antes de exportar, se utilicen los valores actuales.

# Importante sobre `RowsCopy`

El bloque que aparece comentado en la imagen:

```powerscript
RowsCopy(...)
```

no es necesario para esta solución.

Como el DataStore recuperará directamente los datos mediante el mismo filtro de tipo `8`, es mejor:

```text
Retrieve del DataStore
→ SaveAs del DataStore
```

en lugar de:

```text
RowsCopy del DataWindow de nueve columnas
→ conversión manual de columnas
```

Eso evita problemas de:

- nombres de columnas;
- tipos numéricos;
- valores nulos;
- conversiones de decimal;
- cantidad de registros.

Una vez agregues las declaraciones y la reconstrucción de los filtros, los tres warnings de la imagen deben desaparecer.
