/*
PUNTO 3
Detección de lugares de atención vigentes relacionados
con servicios vigentes cuya especialidad está dada de baja.

SET NOCOUNT ON

SELECT
    prestador_codigo            = PL.prestad,
    prestador_nombre            = P.ape_razon,
    lugar_codigo                = PL.lugar,
    lugar_nombre                = PL.nombre,
    lugar_domicilio             = PL.domi,
    lugar_vigencia_desde        = PL.vigen_desde,
    lugar_baja_fecha            = PL.baja_fecha,
    servicio_tipo               = PS.servi_tipo,
    especialidad_codigo         = PS.espe,
    especialidad_descripcion    = E.exten_deno,
    servicio_alta_fecha         = PS.alta_fecha,
    servicio_baja_fecha         = PS.baja_fecha,
    especialidad_alta_fecha     = E.alta_fecha,
    especialidad_baja_fecha     = E.baja_fecha
FROM dbo.PRESTAD_LUGARES PL

     INNER JOIN dbo.PRESTADORES P
     ON P.prestad = PL.prestad

     INNER JOIN dbo.PRESTAD_SERVICIOS PS
     ON PS.prestad = PL.prestad
     AND PS.lugar = PL.lugar

     INNER JOIN dbo.ESPECIALIDADES E
     ON E.espe = PS.espe

WHERE PL.baja_fecha IS NULL
AND PS.baja_fecha IS NULL
AND E.baja_fecha IS NOT NULL

ORDER BY
    PL.prestad,
    PL.lugar,
    PS.espe,
    PS.servi_tipo
