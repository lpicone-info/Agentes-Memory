# 3. Calcular el total esperado por agrupación

```sql
DECLARE @fecha_ejecucion DATETIME = GETDATE();

;WITH VIGENTE AS
(
    SELECT
        CP.conve,
        CP.nomen,
        RTRIM(CP.prestac) AS prestac,
        CP.concep,
        CP.conve_prestac,
        V.vigen_desde,
        V.valor,
        V.vier_plus

    JOIN CONVE_PRESTAC_VALORES V
      ON V.conve_prestac = CP.conve_prestac
    WHERE CP.conve IN (329, 343, 2387)
      AND CP.nomen = 15
      AND CP.prestac = '811207'
      AND CP.baja_fecha IS NULL
      AND V.baja_fecha IS NULL
      AND V.vigen_desde =
          (
              SELECT MAX(V2.vigen_desde)
              FROM CONVE_PRESTAC_VALORES V2
              WHERE V2.conve_prestac = V.conve_prestac
                AND V2.baja_fecha IS NULL
                AND V2.vigen_desde <= @fecha_ejecucion
          )
)
SELECT
    conve,
    nomen,
    prestac,
    COUNT(*) AS CONCEPTOS_VIGENTES,
    SUM(CASE WHEN valor > 0.01 THEN 1 ELSE 0 END)
        AS CONCEPTOS_INCLUIDOS,
    CAST(
        SUM(
            CASE
                WHEN valor > 0.01
                    THEN valor + ISNULL(vier_plus, 0)
                ELSE 0
            END
        )
        AS DECIMAL(14,2)
    ) AS TOTAL_ESPERADO,
    MAX(
        CASE
            WHEN valor > 0.01 THEN vigen_desde
        END
    ) AS VIGENCIA_ESPERADA
FROM VIGENTE
GROUP BY
    conve,
    nomen,
    prestac
ORDER BY conve;
```
