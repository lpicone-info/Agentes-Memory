GO

DECLARE @kprestad INTEGER = 4732;

SELECT
    lugar = PS.lugar,

    servi =
          CONVERT(CHAR(5), PS.servi_tipo)
        + CONVERT(CHAR(5), PS.espe)
        + CONVERT(CHAR(5), PS.subespe),

    deno =
        CASE
            WHEN SUBE.exten_deno IS NULL
                THEN ST.deno + '/ ' + E.exten_deno
            ELSE
                  ST.deno
                + '/ ' + E.exten_deno
                + '/ ' + SUBE.exten_deno
        END

FROM dbo.PRESTAD_SERVICIOS PS

INNER JOIN dbo.ESPECIALIDADES E
    ON E.espe = PS.espe

INNER JOIN dbo.SERVI_TIPOS ST
    ON ST.servi_tipo = PS.servi_tipo

LEFT JOIN dbo.SUBESPECIALIDADES SUBE
    ON SUBE.subespe = PS.subespe

WHERE PS.prestad = @kprestad

  AND ISNULL(PS.baja_fecha, '20500101') > GETDATE()
  AND ISNULL(PS.recu_fecha, '19000101') <= GETDATE()

ORDER BY
    PS.lugar,

    + CONVERT(CHAR(5), PS.espe)
