```sql
SELECT
    A.sucur,
    A.auto,
    A.origen,
    A.esta,
    A.alta_fecha,
    A.baja_fecha,
    A.usuario,
    A.recha,
    A.auto_esta_moti
FROM AUTORIZACIONES A
WHERE A.baja_fecha IS NOT NULL
AND A.auto_esta_moti IS NULL
AND EXISTS (
    SELECT 1
    FROM AUTO_AUDITORIA AA
    WHERE AA.sucur = A.sucur
    AND AA.auto = A.auto
    AND AA.activi = 'Retiro de Autorización'
)
AND EXISTS (
    SELECT 1
    FROM AUTO_AUDITORIA AA
    WHERE AA.sucur = A.sucur
    AND AA.auto = A.auto
    AND AA.activi = 'Reversión estado autorización'
)
ORDER BY A.baja_fecha, A.sucur, A.auto
```

Si necesitás ver también las fechas de ambas auditorías:

```sql
SELECT
    A.sucur,
    A.auto,
    A.origen,
    A.esta,
    A.alta_fecha,
    A.baja_fecha,
    A.usuario,
    A.recha,
    A.auto_esta_moti,
    retiro_fecha = MAX(CASE WHEN AA.activi = 'Retiro de Autorización' THEN AA.fecha_hora END),
    reversion_fecha = MAX(CASE WHEN AA.activi = 'Reversión estado autorización' THEN AA.fecha_hora END)
FROM AUTORIZACIONES A
INNER JOIN AUTO_AUDITORIA AA
    ON AA.sucur = A.sucur
    AND AA.auto = A.auto
WHERE A.baja_fecha IS NOT NULL
AND A.auto_esta_moti IS NULL
AND AA.activi IN (
    'Retiro de Autorización',
    'Reversión estado autorización'
)
GROUP BY
    A.sucur,
    A.auto,
    A.origen,
    A.esta,
    A.alta_fecha,
    A.baja_fecha,
    A.usuario,
    A.recha,
    A.auto_esta_moti
HAVING COUNT(DISTINCT AA.activi) = 2
ORDER BY A.baja_fecha, A.sucur, A.auto
```
