public function long wf_texttoarray_utf8 (string as_archivo, ref string as_vector[], long al_ini, long al_fin)

LONG    ll_indice, ll_linea
INTEGER li_archivo, li_resultado
STRING  ls_linea, ls_vector_nulo[]

IF IsNull(al_fin) THEN
    al_fin = 0
END IF

as_vector = ls_vector_nulo

IF NOT FileExists(as_archivo) THEN
    RETURN -1
END IF

li_archivo = FileOpen( &
    as_archivo, &
    LineMode!, &
    Read!, &
    LockReadWrite!, &
    Replace!, &
    EncodingUTF8! &
)

IF li_archivo = -1 THEN
    RETURN -2
END IF

li_resultado = 0
ll_indice    = 0
ll_linea     = 0

DO WHILE li_resultado >= 0

    li_resultado = FileRead(li_archivo, ls_linea)
    ll_linea ++

    IF li_resultado = -1 THEN

        FileClose(li_archivo)
        RETURN -1

    ELSEIF li_resultado > 0 THEN

        IF al_fin = 0 OR ll_linea >= al_ini THEN
            ll_indice ++
            as_vector[ll_indice] = ls_linea
        END IF

    END IF

    IF al_fin > 0 AND ll_linea >= al_fin THEN
        EXIT
    END IF

LOOP

FileClose(li_archivo)

RETURN ll_indice

end function
