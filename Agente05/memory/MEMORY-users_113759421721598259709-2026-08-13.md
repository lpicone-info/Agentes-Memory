LONG ll_row

DECLARE soli_auto_esta_actual PROCEDURE FOR dbo.prs_solicitudes_auto_esta_actual
    @prm_soli_auto = :idc_soli_auto_refrescar,
    @prm_echo      = 'S';

    CLOSE soli_auto_esta_actual
    RETURN
END IF

FETCH soli_auto_esta_actual INTO :ls_soli_esta_actual;

IF SQLCA.SQLCode <> 0 THEN

    RETURN
END IF

CLOSE soli_auto_esta_actual

ll_row = dw_lista.Find("soli_auto = " + STRING(idc_soli_auto_refrescar), 1, dw_lista.RowCount())
