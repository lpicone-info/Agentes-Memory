## Solución recomendada

Crear un nuevo SP específico para la solapa **Rango genérico**, reutilizando los mismos filtros y joins:

```sql
CREATE PROCEDURE dbo.prs_tmp_rango_generico_cobertura_export
(
    @prm_prepaga      SMALLINT,
    @prm_vigen_desde  DATETIME,
    @prm_planes       XML
)
AS
BEGIN

    SET NOCOUNT ON
    SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

    CREATE TABLE #PLANES
    (
        plan_codi CHAR(10) NOT NULL
    )

    INSERT INTO #PLANES
    (
        plan_codi
    )
    SELECT
        plan_codi = N.value('.', 'CHAR(10)')
    FROM @prm_planes.nodes('x') AS T(N)

    SELECT DISTINCT
        id_rango_valores = CCG.cod,
        rango_id         = CCGRV.rango_id,
        valor            = CONVERT(NUMERIC(12), CCGRV.valor)
    FROM COBER_COPA_GRUPOS AS CCG
    INNER JOIN PLANES AS P
        ON P.plan_codi = CCG.plan_codi
    INNER JOIN #PLANES AS T
        ON T.plan_codi = CCG.plan_codi
    INNER JOIN COBER_COPA_GRUPOS_RANGO_VALOR AS CCGRV
        ON CCGRV.cod_grupo = CCG.cod
    WHERE CCG.baja_fecha IS NULL
      AND P.prepaga = @prm_prepaga
      AND CCG.vigen_desde >= @prm_vigen_desde
      AND CCG.tipo_grupo = 'S'
      AND CCG.prestad IS NULL
      AND CCGRV.baja_fecha IS NULL
    ORDER BY
        CCG.cod,
        CCGRV.rango_id

    DROP TABLE #PLANES

END
```
