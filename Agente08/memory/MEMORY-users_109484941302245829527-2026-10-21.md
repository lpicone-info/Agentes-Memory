```sql
WITH documentos_duplicados AS
(
    SELECT
        prepaga,
        contra,
        docu_tipo,
        docu_nro,
        COUNT(*) AS cantidad_registros
    FROM afiliados
    WHERE docu_tipo IS NOT NULL
      AND ISNULL(docu_nro, 0) <> 0
    GROUP BY
        prepaga,
        contra,
        docu_tipo,
        docu_nro
    HAVING COUNT(*) > 1
)
SELECT
    a.prepaga AS entidad,
    a.contra AS contrato,
    a.inte AS integrante,
    a.paren AS parentesco,
    a.ape AS apellido,
    a.nombre,
    a.docu_tipo AS tipo_documento,
    a.docu_nro AS numero_documento,
    a.ingre_fecha AS fecha_ingreso,
    a.baja_fecha AS fecha_baja,
    CASE
        WHEN a.baja_fecha IS NULL THEN 'ACTIVO'
        ELSE 'DADO DE BAJA'
    END AS estado,
    d.cantidad_registros
FROM afiliados a
JOIN documentos_duplicados d
    ON d.prepaga = a.prepaga
   AND d.contra = a.contra
   AND d.docu_tipo = a.docu_tipo
   AND d.docu_nro = a.docu_nro
ORDER BY
    a.prepaga,
    a.contra,
    a.docu_tipo,
    a.docu_nro,
    a.inte
```
