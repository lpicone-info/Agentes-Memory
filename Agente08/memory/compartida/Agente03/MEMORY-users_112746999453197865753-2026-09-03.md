Función SQL Server propuesta:

```sql
CREATE FUNCTION dbo.afn_code25i_encode
(
    @prm_valor VARCHAR(100)
)
RETURNS VARCHAR(4000)
AS
BEGIN
    DECLARE @var_valor VARCHAR(100)
    DECLARE @var_salida VARCHAR(4000)
    DECLARE @var_pos INT
    DECLARE @var_par INT
    DECLARE @var_len INT

    SET @var_valor = REPLACE(@prm_valor, ' ', '')

    IF @var_valor IS NULL RETURN NULL
    IF @var_valor = '' RETURN NULL
    IF PATINDEX('%[^0-9]%', @var_valor) > 0 RETURN NULL

    IF LEN(@var_valor) % 2 = 1
        SET @var_valor = '0' + @var_valor

    SET @var_salida = CHAR(201)
    SET @var_pos = 1
    SET @var_len = LEN(@var_valor)

    WHILE @var_pos <= @var_len
    BEGIN
        SET @var_par = CONVERT(INT, SUBSTRING(@var_valor, @var_pos, 2))

        IF @var_par <= 94
            SET @var_salida = @var_salida + CHAR(@var_par + 32)
        ELSE
            SET @var_salida = @var_salida + CHAR(@var_par + 100)

        SET @var_pos = @var_pos + 2
    END

    SET @var_salida = @var_salida + CHAR(202)

    RETURN @var_salida
END
GO
```
