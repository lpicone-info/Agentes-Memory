**SP nuevo**

```sql
CREATE PROCEDURE dbo.afs_programacion_facturacion_pendiente (
    @prm_prepaga        smallint,
    @prm_emi_grupo      tinyint,
    @prm_factu_periodo  date,
    @prm_tipo_proceso   char(6)
) AS
BEGIN
    SET NOCOUNT ON

    SELECT TOP 1
            existe_programacion = 1,
            fecha_programada    = pa.fecha_programada,
            hora_programada     = CONVERT(char(5), pa.hora_programada),
            id                  = pa.id
    FROM    dbo.PROGRAMACION_FACTURACION as pa
    WHERE   pa.prepaga = @prm_prepaga
    AND     pa.emi_grupo = @prm_emi_grupo
    AND     pa.factu_perio = @prm_factu_periodo
    AND     pa.tipo_proceso = @prm_tipo_proceso
    AND     pa.fecha_baja IS NULL
    AND     pa.fecha_ejecucion IS NULL
    ORDER
    BY      pa.fecha_programada ASC,
            pa.hora_programada ASC,
            pa.id ASC
END
GO
```
