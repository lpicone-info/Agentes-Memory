Podés usar esta consulta en SQL Server. Devuelve prestadores activos que:

- tienen convenio vigente asociado;
- tienen más de una categoría vigente;
- informa el tipo (`C = Círculo`, `P = Prestador`, etc.);
- muestra las categorías y convenios asociados.

```sql
WITH categorias AS
(
    SELECT
           pch.prestad,
           COUNT(DISTINCT pch.cate) AS cantidad_categorias,
           STUFF
           (
               (
                   SELECT DISTINCT
                          ', ' + CAST(pch2.cate AS VARCHAR(10))
                   FROM prestad_cate_habi pch2
                   WHERE pch2.prestad = pch.prestad
                     AND pch2.baja_fecha IS NULL
                   FOR XML PATH(''), TYPE
               ).value('.', 'VARCHAR(MAX)'),
               1,
               2,
               ''
           ) AS categorias
    FROM prestad_cate_habi pch
    WHERE pch.baja_fecha IS NULL
    GROUP BY pch.prestad
),
convenios AS
(
    SELECT
           cc.prestad,
           COUNT(DISTINCT cc.conve) AS cantidad_convenios,
           STUFF
           (
               (
                   SELECT DISTINCT
                          ', ' + CAST(cc2.conve AS VARCHAR(10))
                   FROM carti_convenios cc2
                   INNER JOIN convenios c2
                           ON c2.conve = cc2.conve
                          AND c2.baja_fecha IS NULL
                   WHERE cc2.prestad = cc.prestad
                     AND cc2.baja_fecha IS NULL
                   FOR XML PATH(''), TYPE
               ).value('.', 'VARCHAR(MAX)'),
               1,
               2,
               ''
           ) AS convenios
    FROM carti_convenios cc
    INNER JOIN convenios c
            ON c.conve = cc.conve
           AND c.baja_fecha IS NULL
    WHERE cc.baja_fecha IS NULL
    GROUP BY cc.prestad
)
SELECT
       p.prestad,
       p.ape_razon AS prestador,
       p.tipo AS tipo_codigo,
       CASE p.tipo
           WHEN 'C' THEN 'Círculo'
           WHEN 'P' THEN 'Prestador'
           WHEN 'F' THEN 'Farmacia'
           WHEN 'S' THEN 'Sanatorio'
           ELSE 'Otro'
       END AS tipo_descripcion,
       cat.cantidad_categorias,
       cat.categorias,
       conv.cantidad_convenios,
       conv.convenios
FROM prestadores p
INNER JOIN categorias cat
        ON cat.prestad = p.prestad
INNER JOIN convenios conv
        ON conv.prestad = p.prestad
WHERE p.baja_fecha IS NULL
  AND cat.cantidad_categorias > 1
ORDER BY
       p.tipo,
       p.ape_razon;
```
