DECLARE @Prepaga SMALLINT
DECLARE @FechaDesde CHAR(8)
DECLARE @FechaHasta CHAR(8)

SET @Prepaga = 1
SET @FechaDesde = '20250101'
SET @FechaHasta = '20251231'

SELECT
    t.line_nro,
    contrato = LEFT(LTRIM(t.contra), LEN(LTRIM(t.contra))),
    integrante = LEFT(LTRIM(t.inte), LEN(LTRIM(t.inte))),
    apellido = LEFT(LTRIM(t.ape), LEN(LTRIM(t.ape))),
    nombre = LEFT(LTRIM(t.nombre), LEN(LTRIM(t.nombre))),
    cuil = LEFT(LTRIM(t.cuil), LEN(LTRIM(t.cuil))),
    fecha_baja_subedatos =
        CASE
            WHEN ISDATE(LTRIM(t.baja_fecha)) = 1
            THEN CONVERT(DATETIME, LTRIM(t.baja_fecha), 112)
        END,
    motivo_baja_subedatos =
        CASE
            WHEN ISNUMERIC(t.baja_moti) = 1
            THEN CONVERT(SMALLINT, t.baja_moti)
        END,
    motivo_descripcion = bm.deno,
    fecha_baja_afiliado = a.baja_fecha,
    motivo_baja_afiliado = a.baja_moti,
    estado_aplicacion =
        CASE
            WHEN a.contra IS NULL
                THEN 'AFILIADO NO ENCONTRADO'
            WHEN a.baja_fecha IS NULL
                THEN 'BAJA NO APLICADA'
            WHEN CONVERT(CHAR(8), a.baja_fecha, 112) = LTRIM(t.baja_fecha)
             AND a.baja_moti = CONVERT(SMALLINT, t.baja_moti)
                THEN 'BAJA APLICADA Y COINCIDENTE'
            ELSE 'BAJA APLICADA CON DIFERENCIAS'
        END
FROM TEMP_SUBEDATOS_AFILIADOS t

LEFT JOIN AFILIADOS a
    ON  a.prepaga = @Prepaga
    AND a.contra = t.contra
    AND a.inte = t.inte

LEFT JOIN BAJA_MOTIVOS bm
    ON bm.baja_moti =
        CASE
            WHEN ISNUMERIC(t.baja_moti) = 1
            THEN CONVERT(SMALLINT, t.baja_moti)
        END

WHERE NULLIF(LTRIM(t.baja_fecha), '') IS NOT NULL
AND ISDATE(LTRIM(t.baja_fecha)) = 1
AND (@FechaDesde IS NULL OR LTRIM(t.baja_fecha) >= @FechaDesde)
AND (@FechaHasta IS NULL OR LTRIM(t.baja_fecha) <= @FechaHasta)

ORDER BY
    LTRIM(t.baja_fecha) DESC,
    t.line_nro
