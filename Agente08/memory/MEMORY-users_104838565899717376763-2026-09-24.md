  - `liqui_fecha IS NULL`
  - `baja_fecha IS NULL`

### Consulta SQL

```sql
SELECT TOP 20
       e.ofi,
       e.expe,
       et.deno AS tipo_expediente,
       e.perio,
       e.fecha,
       e.recep_fecha,
       e.cierre_admin_fecha,
       e.prestad,
       p.ape_razon,
       p.nombre_abre,
       COUNT(DISTINCT eo.item) AS ordenes_odontologicas,
       COUNT(eoi.orden) AS practicas_odontologicas
FROM expedientes e
INNER JOIN expe_tipos et
        ON et.tipo = e.tipo
       AND et.baja_fecha IS NULL
INNER JOIN prestadores p
        ON p.prestad = e.prestad
INNER JOIN expe_odon eo
        ON eo.ofi = e.ofi
       AND eo.expe = e.expe
       AND eo.baja_fecha IS NULL
LEFT JOIN expe_odon_items eoi
       ON eoi.ofi = eo.ofi
      AND eoi.expe = eo.expe
      AND eoi.item = eo.item
      AND eoi.baja_fecha IS NULL
WHERE e.tipo = 'D'
  AND e.baja_fecha IS NULL
  AND e.cierre_admin_fecha IS NOT NULL
  AND e.super_fecha IS NULL
  AND e.liqui_fecha IS NULL
GROUP BY e.ofi,
         e.expe,
         et.deno,
         e.perio,
         e.fecha,
         e.recep_fecha,
         e.cierre_admin_fecha,
         e.prestad,
         p.ape_razon,
         p.nombre_abre
ORDER BY e.cierre_admin_fecha DESC,
         e.ofi DESC,
         e.expe DESC
```
