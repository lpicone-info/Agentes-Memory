(

        prepaga,
        plan_codi,
        tari

prefacturas_afectadas AS

        p.plan_codi,

        COUNT(*) AS lineas_afectadas,
        SUM(p.impor) AS importe_con_1_70,
        SUM(
            ROUND(
                p.impor / 1.017 * 1.019,
                2
            )
        ) AS importe_estimado_1_90
    FROM prefacturas p
    INNER JOIN combinaciones_afectadas ca
            ON ca.prepaga = p.prepaga

    WHERE p.factu_perio = '20261201'
      AND p.factu_concep = 'CUO'
      AND p.baja_fecha IS NULL
    GROUP BY
        p.prepaga,
        p.contra,
        p.plan_codi,
        p.tari
)
SELECT
    c.ente,
    c.compro_tipo,
    c.compro_nro,
    c.contra,
    a.ape,
    a.nombre,
    pa.plan_codi,
    pl.deno AS denominacion_plan,
    pa.tari,
    t.deno AS denominacion_tarifa,
    pa.lineas_afectadas,
    pa.importe_con_1_70,
    pa.importe_estimado_1_90,

FROM comprobantes c
INNER JOIN prefacturas_afectadas pa
