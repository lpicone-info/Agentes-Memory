			--SELECT  @var_efe_lugar = MIN(lugar)

  ELSE

	IF (@tipo_transac = '02Q')
  BEGIN
		SET @efe_circu_sana = @var_cate		
	END

  IF @tipo = 'I' AND @aufree IS NULL
  BEGIN
    SET @prm_interTipo    = @carac
