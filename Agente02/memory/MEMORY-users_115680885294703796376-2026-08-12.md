  "Ya existe una derivación vigente para la misma Obra Social y CUIT",

      if (!d.os || !d.cuit || !d.vigenDesde) return false;

      const desde = new Date(d.vigenDesde);
      desde.setHours(0, 0, 0, 0);

      const hasta = d.vigenHasta ? new Date(d.vigenHasta) : null;
      if (hasta) hasta.setHours(0, 0, 0, 0);

      return desde <= apliFecha && (!hasta || hasta >= apliFecha);
    });
