event type integer ue_validar_fecha(datetime ad_fecha, string as_colname);/*Realiza la validación de las fechas de la autorización
	ad_fecha			Fecha a validar
	as_colname		Columna				*/
DATETIME ldt_orden_fecha, ldt_fecha, ldt_vali_fecha, ld_null
DATASTORE lds_param_auto
INTEGER li_dias_auto_orden

lds_param_auto = CREATE DATASTORE
lds_param_auto.dataobject = 'dw_param_auto'
lds_param_auto.SetTransObject(sqlca)
lds_param_auto.Retrieve()
IF lds_param_auto.Rowcount() > 0 THEN
	li_dias_auto_orden = lds_param_auto.object.dias_auto_orden[1]     // 8009 20140205
END IF
DESTROY lds_param_auto

//SetNull(ld_null)
SetNull (ld_null )
//Valido el tipo de dato
IF NOT lib.of_es_smalldatetime_valido(ad_fecha) THEN
	mbox(THIS,"Ingrese una fecha válida.",1,0)
	RETURN 1
END IF

//Valido que haya ingresado una fecha
CHOOSE CASE as_colname
	CASE 'fecha'	
		IF lib.of_vacio(ad_fecha) THEN 
			mbox(THIS,"Ingrese la fecha de autorización",1,0)
			RETURN 1
			
		END IF
		
	CASE 'orden_fecha'
		IF lib.of_vacio(ad_fecha) THEN 
			mbox(THIS,"Ingrese la fecha de la orden",1,0)
			dw_auto.SetColumn('orden_fecha')
			SetFocus()
			RETURN 1
		END IF
		
	CASE 'vali_fecha'
		IF IsNull(ad_fecha) THEN
			mbox(THIS,"Ingrese la fecha de validez de la autorización",1,0)
			RETURN 1
		END IF
END CHOOSE		

//Valido las fechas entre si
ldt_fecha 			= dw_auto.GetItemDateTime(1,'fecha')
ldt_orden_fecha	= dw_auto.GetItemDateTime(1,'orden_fecha')
ldt_vali_fecha 	= dw_auto.GetItemDateTime(1,'vali_fecha')

IF NOT lib.of_vacio(ldt_orden_fecha) THEN 
	IF ldt_orden_fecha > ldt_fecha THEN
		mbox(THIS,"La fecha de la orden no puede ser mayor a la fecha de la autorización",1,0)
		
		dw_auto.object.orden_fecha[1] = ld_null	
		dw_auto.SetColumn('orden_fecha')
		dw_auto.SetFocus()
		RETURN 1
	END IF
	
	/*SF 8009 - validación de antiguedad permitida para la fecha de la orden*/
	IF ldt_orden_fecha < datetime(RelativeDate(date(ldt_fecha), -li_dias_auto_orden)) THEN
		mbox(THIS,"La fecha de la orden supera los "+STRING(li_dias_auto_orden)+" días de antigüedad de la fecha de la autorización.",1,0)
		
		dw_auto.object.orden_fecha[1] = ld_null	
		dw_auto.SetColumn('orden_fecha')
		dw_auto.SetFocus()
		RETURN 1
	END IF
	
END IF

IF NOT lib.of_vacio(ldt_orden_fecha) THEN 
	IF ldt_orden_fecha > ldt_vali_fecha THEN
		mbox(THIS,"La fecha de la orden no puede ser mayor a la fecha de validez",1,0)
		
		dw_auto.object.orden_fecha[1] = ld_null	
		dw_auto.SetColumn('orden_fecha')
		dw_auto.SetFocus()
		RETURN 1
	END IF	
END IF

IF NOT lib.of_vacio(ldt_vali_fecha) THEN 
	IF ldt_vali_fecha < ldt_fecha THEN
		mbox(THIS,"La fecha de validez no puede ser menor a la fecha de la autorización",1,0)
		dw_auto.object.vali_fecha[1] = ld_null	
		dw_auto.SetColumn('vali_fecha')
		dw_auto.SetFocus()		
		RETURN 1
	END IF	
END IF

RETURN 0
end event

 //Averiguo el nivel de la prestación

	IF li_nivel = 1 AND  gi_usuari_nivel < 1 THEN
