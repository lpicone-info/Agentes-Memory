```js
'use strict'

require('dotenv').config()

const fs = require('fs')
const path = require('path')
const express = require('express')
const soap = require('soap')
const { precargaSolicitudes } = require('./services/precargaSolicitudes')
const { closePool } = require('./db')

const app = express()
const port = Number.parseInt(process.env.PORT || '3000', 10)
const soapPath = process.env.SOAP_PATH || '/PrecargaSolicitudes'
const wsdlPath = path.join(__dirname, 'soap', 'precargaSolicitudes.wsdl')
const wsdl = fs.readFileSync(wsdlPath, 'utf8')

const service = {
  PrecargaSolicitudesService: {
    PrecargaSolicitudesPort: {
      PrecargaSolicitudes: precargaSolicitudes
    }
  }
}

app.get('/health', (_req, res) => {
  res.json({ status: 'ok', service: 'PrecargaSolicitudes' })
})

const server = app.listen(port, () => {
  soap.listen(app, soapPath, service, wsdl)
  console.log(`PrecargaSolicitudes SOAP listening on http://localhost:${port}${soapPath}`)
})

async function shutdown(signal) {
  console.log(`${signal} received. Closing server...`)
  server.close(async () => {
    await closePool()
    process.exit(0)
  })
}

process.on('SIGINT', () => shutdown('SIGINT'))
process.on('SIGTERM', () => shutdown('SIGTERM'))
```

```js
'use strict'

const sql = require('mssql')

let poolPromise = null

function boolFromEnv(value, defaultValue) {
  if (value === undefined || value === null || value === '') return defaultValue
  return String(value).toLowerCase() === 'true'
}

function intFromEnv(value, defaultValue) {
  const parsed = Number.parseInt(value, 10)
  return Number.isFinite(parsed) ? parsed : defaultValue
}

function getConfig() {
  return {
    server: process.env.DB_SERVER,
    port: intFromEnv(process.env.DB_PORT, 1433),
    database: process.env.DB_DATABASE,
    user: process.env.DB_USER,
    password: process.env.DB_PASSWORD,
    requestTimeout: intFromEnv(process.env.DB_REQUEST_TIMEOUT_MS, 30000),
    pool: {
      max: intFromEnv(process.env.DB_POOL_MAX, 10),
      min: intFromEnv(process.env.DB_POOL_MIN, 0),
      idleTimeoutMillis: intFromEnv(process.env.DB_POOL_IDLE_TIMEOUT_MS, 30000)
    },
    options: {
      encrypt: boolFromEnv(process.env.DB_ENCRYPT, false),
      trustServerCertificate: boolFromEnv(process.env.DB_TRUST_SERVER_CERTIFICATE, true)
    }
  }
}

async function getPool() {
  if (!poolPromise) {
    poolPromise = sql.connect(getConfig())
  }
  return poolPromise
}

async function closePool() {
  if (poolPromise) {
    const pool = await poolPromise
    poolPromise = null
    await pool.close()
  }
}

module.exports = {
  sql,
  getPool,
  closePool
}
```

### `src/services/precargaSolicitudes.js`

```js
'use strict'

const { sql, getPool } = require('../db')

async function precargaSolicitudes(args) {
  const mensaje = args && Object.prototype.hasOwnProperty.call(args, 'mensaje')
    ? args.mensaje
    : ''

  const pool = await getPool()
  const result = await pool.request()
    .input('mensaje', sql.NVarChar(sql.MAX), mensaje)
    .execute('ars_precarga_solicitud')

  const row = result.recordset && result.recordset.length > 0
    ? result.recordset[0]
    : null

  return {
    return: row && row.out_xml !== undefined && row.out_xml !== null
      ? String(row.out_xml)
      : ''
  }
}

module.exports = {
  precargaSolicitudes
}
```
