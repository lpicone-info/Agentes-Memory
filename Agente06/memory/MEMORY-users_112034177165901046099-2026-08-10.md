    @id_reintegro INT,
    @resultado INT OUTPUT,
    @mensaje VARCHAR(255) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    SET @resultado = 0;
    SET @mensaje = '';

    IF EXISTS (
        SELECT 1
        FROM dbo.reintegros
        WHERE id_reintegro = @id_reintegro
          AND id_lote IS NOT NULL
    )
    BEGIN
        SET @resultado = 1001;
        SET @mensaje = 'El reintegro seleccionado ya pertenece a otro lote.';
        RETURN;
    END

    -- resto del proceso normal

    SET @resultado = 0;
    SET @mensaje = 'OK';
END
