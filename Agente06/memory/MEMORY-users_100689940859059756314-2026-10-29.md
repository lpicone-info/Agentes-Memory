### 1. `pri_expe_transac_remotas_hono.sql`

Debe contar los lugares vigentes cuando el lugar original esté dado de baja:

```sql
DECLARE @cantidad_lugares_vigentes int

IF NOT EXISTS (
    SELECT 1
    FROM prestad_lugares
    WHERE prestad = @efe_prestad
      AND lugar = @lugar
      AND baja_fecha IS NULL
)
BEGIN
    SELECT @cantidad_lugares_vigentes = COUNT(DISTINCT lugar)
    FROM prestad_lugares
    WHERE prestad = @efe_prestad
      AND baja_fecha IS NULL

    IF @cantidad_lugares_vigentes = 1
    BEGIN
        SELECT @lugar = MIN(lugar)
        FROM prestad_lugares
        WHERE prestad = @efe_prestad
          AND baja_fecha IS NULL
    END
    ELSE
    BEGIN
        IF @cantidad_lugares_vigentes = 0
        BEGIN
            SET @mensaje_error =
                'No se puede importar la transacción. El lugar de prestación informado está dado de baja y el prestador no posee lugares vigentes.'
        END
        ELSE
        BEGIN
            SET @mensaje_error =
                'No se puede importar la transacción. El lugar de prestación informado está dado de baja y el prestador posee más de un lugar vigente. No es posible determinar automáticamente el lugar correcto.'

    END
```
