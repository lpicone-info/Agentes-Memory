SELECT
    a.prepaga,
    p.exten_deno AS prepaga_deno,
    LTRIM(RTRIM(a.contra)) AS contra,
    LTRIM(RTRIM(a.inte)) AS inte,
    a.nombre,

    pa.deno AS paren,

    z.deno AS zona_deno,

        WHEN propia.tipo_aportante = 'T' THEN 'Si'

    END AS deriva,

    LTRIM(RTRIM(propia.inte_titular_aportante))

FROM AFILIADOS a

INNER JOIN PREPAGAS p

LEFT JOIN PARENTESCOS pa

-- Determina el tipo de aportante del integrante evaluado
OUTER APPLY

        d.tipo_aportante,
        d.inte_titular_aportante
    FROM AFI_DERIV_APORTES d

      AND d.inte = a.inte

            d.baja_fecha IS NULL
            OR d.baja_fecha > @fecha
          )
