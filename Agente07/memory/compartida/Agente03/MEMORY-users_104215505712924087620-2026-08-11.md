CREATE PROCEDURE dbo.afs_programacion_facturacion_duplicada (
    @prm_tipo_proceso       char(6),
    @prm_prepaga            smallint,
    @prm_emi_grupo          tinyint,
    @prm_factu_perio        date,
    @prm_fecha_programada   date,
    @prm_hora_programada    time
) AS
BEGIN
    SET NOCOUNT ON

    SELECT existe = CONVERT(integer,
        CASE
            WHEN EXISTS (
                SELECT  1
                FROM    dbo.PROGRAMACION_FACTURACION
                WHERE   tipo_proceso = @prm_tipo_proceso
                AND     prepaga = @prm_prepaga
                AND     emi_grupo = @prm_emi_grupo
                AND     factu_perio = @prm_factu_perio
                AND     fecha_programada = @prm_fecha_programada

END
GO
