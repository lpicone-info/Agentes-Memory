```sql
SET NOCOUNT ON

DECLARE @prepaga smallint = 1
DECLARE @contra char(15) = '8673'
DECLARE @inte char(2) = '01'
DECLARE @inte_titular char(2) = '00'
DECLARE @vigen_desde smalldatetime = '20260501'

SELECT 'ANTES' AS estado, *
FROM afi_deriv_aportes
WHERE prepaga = @prepaga
AND contra = @contra
AND inte IN (@inte_titular, @inte)
ORDER BY inte, vigen_desde

BEGIN TRAN

INSERT INTO afi_deriv_aportes (
    prepaga, contra, inte, vigen_desde, obra_soc, empresa, legajo,
    sueldo, deriva, alta_fecha, baja_fecha, cuit, vigen_hasta,
    meses_a_estimar, inte_titular, sistema, fecha_act_sueldo,
    fecha_desde, fecha_hasta, verificacion_aporte, aporte_esperado,
    nro_sss, tipo_aportante, inte_titular_aportante
)
SELECT TOP 1
    t.prepaga, t.contra, @inte, @vigen_desde, t.obra_soc, t.empresa,
    t.legajo, 0, 'N', GETDATE(), NULL, t.cuit, NULL,
    t.meses_a_estimar, t.inte_titular, t.sistema, t.fecha_act_sueldo,
    t.fecha_desde, t.fecha_hasta, t.verificacion_aporte,
    t.aporte_esperado, t.nro_sss, 'G', @inte_titular
FROM afi_deriv_aportes t
INNER JOIN afiliados a
    ON a.prepaga = t.prepaga
    AND a.contra = t.contra
    AND a.inte = @inte
    AND a.baja_fecha IS NULL
    AND a.paren <> 'T'
WHERE t.prepaga = @prepaga
AND t.contra = @contra
AND t.inte = @inte_titular
AND t.baja_fecha IS NULL
AND EXISTS (
    SELECT 1
    FROM afi_opcion o
    WHERE o.prepaga = t.prepaga
    AND o.contra = t.contra
    AND o.inte = t.inte
    AND o.baja_fecha IS NULL
)
AND NOT EXISTS (
    SELECT 1
    FROM afi_deriv_aportes x
    WHERE x.prepaga = @prepaga
    AND x.contra = @contra
    AND x.inte = @inte
    AND x.baja_fecha IS NULL
)
ORDER BY t.vigen_desde DESC

SELECT @@ROWCOUNT AS registros_insertados

SELECT 'DESPUES' AS estado, *
FROM afi_deriv_aportes
WHERE prepaga = @prepaga
AND contra = @contra
AND inte IN (@inte_titular, @inte)
ORDER BY inte, vigen_desde

COMMIT
```
