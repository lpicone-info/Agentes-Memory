DECLARE @Prepaga SMALLINT = 1;

DECLARE @Contratos TABLE
(
    contra VARCHAR(15) NOT NULL PRIMARY KEY
);

INSERT INTO @Contratos (contra)
VALUES
    ('27584361');
    --,('OTRO_CONTRATO')

;WITH ComprobantesOrdenados AS
(
    SELECT
        c.*,
        ROW_NUMBER() OVER
        (
            PARTITION BY c.prepaga, RTRIM(c.contra)
            ORDER BY
                c.alta_fecha DESC,
                c.fecha DESC,
                c.compro_nro DESC
        ) AS rn
    FROM comprobantes c
    INNER JOIN @Contratos l
        ON l.contra = RTRIM(c.contra)
    WHERE c.prepaga = @Prepaga
      AND c.baja_fecha IS NULL
),
UltimoComprobante AS
(
    SELECT *
    FROM ComprobantesOrdenados
    WHERE rn = 1
),
ItemsUltimo AS
(
    SELECT
        u.prepaga,
        RTRIM(u.contra) AS contra,
        u.ente,
        u.compro_tipo,
        u.compro_nro,

        SUM(ISNULL(ci.valor, 0)) AS suma_items,

        SUM
        (
            CASE
                WHEN RTRIM(ci.factu_concep) = 'AP'
                    THEN ISNULL(ci.valor, 0)
                ELSE 0
            END
        ) AS item_ap,

        SUM
        (
            CASE
                WHEN RTRIM(ci.factu_concep) <> 'AP'
                    THEN ISNULL(ci.valor, 0)
                ELSE 0
            END
        ) AS total_sin_aporte,

        MAX
        (
            CASE
                WHEN RTRIM(ci.factu_concep) = 'AP'
                    THEN RTRIM(ci.obser)
            END
        ) AS observacion_ap

    FROM UltimoComprobante u
    LEFT JOIN compro_items ci
        ON ci.ente = u.ente
       AND ci.compro_tipo = u.compro_tipo
       AND ci.compro_nro = u.compro_nro
       AND ci.baja_fecha IS NULL
    GROUP BY
        u.prepaga,
        u.contra,
        u.ente,
        u.compro_tipo,
        u.compro_nro
),
AportesMismoPeriodo AS
(
    SELECT
        u.prepaga,
        RTRIM(u.contra) AS contra,

        COUNT(a.aporte) AS cantidad_estimaciones,

        SUM(a.impor_estimado) AS importe_estimado_registrado,

        SUM(a.impor_facturado) AS importe_estimado_facturado

    FROM UltimoComprobante u
    LEFT JOIN aportes a
        ON a.prepaga = u.prepaga
       AND RTRIM(a.contra) = RTRIM(u.contra)

       -- Solo se consulta el período del último comprobante.
       AND a.periodo_aporte = u.factu_perio

       AND a.baja_fecha IS NULL
       AND
       (
           a.tipo_aporte = 'E'
           OR a.origen_dato = 'E'
       )
    GROUP BY
        u.prepaga,
        u.contra
),
CalculoDerivaciones AS
(
    SELECT
        u.prepaga,
        RTRIM(u.contra) AS contra,

        COUNT(d.inte) AS derivaciones_titulares,

        SUM
        (
            CASE
                WHEN ai.codigo IS NOT NULL
                THEN ROUND
                (
                    d.sueldo
                    * (ai.valor_apor + ai.valor_cont)
                    / 100.0,
                    2
                )
            END
        ) AS estimado_recalculado

    FROM UltimoComprobante u
    LEFT JOIN afi_deriv_aportes d
        ON d.prepaga = u.prepaga
       AND RTRIM(d.contra) = RTRIM(u.contra)
       AND d.tipo_aportante = 'T'
       AND d.vigen_desde <= u.factu_perio
       AND
       (
           d.vigen_hasta IS NULL
           OR d.vigen_hasta > u.factu_perio
       )
       AND
       (
           d.baja_fecha IS NULL
           OR d.baja_fecha > u.factu_perio
       )
    OUTER APPLY
    (
        SELECT TOP 1
            i.*
        FROM aporte_indices i
        WHERE i.cuenta = d.obra_soc
          AND d.sueldo BETWEEN i.sueldo_desde
                           AND i.sueldo_hasta
          AND i.vigen_desde <= u.factu_perio
          AND
          (
              i.baja_fecha IS NULL
              OR i.baja_fecha > u.factu_perio
          )
        ORDER BY
            i.vigen_desde DESC,
            i.codigo DESC
    ) ai
    GROUP BY
        u.prepaga,
        u.contra
)
SELECT
    l.contra,

    RTRIM(u.compro_tipo) AS compro_tipo,
    RTRIM(u.compro_nro) AS compro_nro,
    u.alta_fecha AS generado_el,
    u.factu_perio,
    u.fecha AS fecha_comprobante,

    u.total AS total_cabecera,
    it.total_sin_aporte,

    -- Se presenta positivo para facilitar la comparación.
    -it.item_ap AS aporte_descontado_comprobante,

    ap.cantidad_estimaciones,
    ap.importe_estimado_registrado,
    ap.importe_estimado_facturado,

    cd.derivaciones_titulares,
    cd.estimado_recalculado,

    ROUND
    (
        (-it.item_ap) - ap.importe_estimado_registrado,
        2
    ) AS diferencia_vs_aportes,

    ROUND
    (
        (-it.item_ap) - cd.estimado_recalculado,
        2
    ) AS diferencia_vs_indices,

    ROUND
    (
        it.suma_items - u.total,
        2
    ) AS diferencia_items_cabecera,

    it.observacion_ap,

    CASE
        WHEN u.compro_nro IS NULL
            THEN 'SIN COMPROBANTE VIGENTE'

        WHEN ISNULL(it.item_ap, 0) = 0
            THEN 'SIN ITEM AP EN EL ULTIMO COMPROBANTE'

        WHEN ap.cantidad_estimaciones = 0
            THEN 'SIN APORTE ESTIMADO PARA EL PERIODO FACTURADO'

        WHEN cd.estimado_recalculado IS NULL
            THEN 'SIN DERIVACION/INDICE PARA RECALCULAR'

        WHEN ABS
             (
                 ROUND
                 (
                     (-it.item_ap)
                     - ap.importe_estimado_registrado,
                     2
                 )
             ) <= 0.01
         AND ABS
             (
                 ROUND
                 (
                     (-it.item_ap)
                     - cd.estimado_recalculado,
                     2
                 )
             ) <= 0.01
         AND ABS
             (
                 ROUND
                 (
                     it.suma_items - u.total,
                     2
                 )
             ) <= 0.01
            THEN 'OK'

        ELSE 'REVISAR'
    END AS resultado

FROM @Contratos l
LEFT JOIN UltimoComprobante u
    ON RTRIM(u.contra) = l.contra
   AND u.prepaga = @Prepaga
LEFT JOIN ItemsUltimo it
    ON it.prepaga = u.prepaga
   AND it.contra = RTRIM(u.contra)
LEFT JOIN AportesMismoPeriodo ap
    ON ap.prepaga = u.prepaga
   AND ap.contra = RTRIM(u.contra)
LEFT JOIN CalculoDerivaciones cd
    ON cd.prepaga = u.prepaga
   AND cd.contra = RTRIM(u.contra)
ORDER BY
    l.contra;
