```sql
WITH parametros AS
(
    SELECT
        prepaga = CONVERT(smallint, 2),
        cuenta_os = CONVERT(int, 547),
        fecha_desde = CONVERT(date, '20250101'),
        fecha_hasta = CONVERT(date, '20261001'),
        fecha_calculo = CONVERT(date, '20261120')
),
ddjj AS
(
    SELECT
        ap.prepaga,
        cuit = ap.cuit,
        cuenta = ISNULL(ap.cuenta, 0),
        periodo = CONVERT(date, ap.periodo),
        ddjj_bruta =
            SUM(
                ISNULL(ap.aporte_calc, 0) +
                ISNULL(ap.cont_calc, 0)
            )
    FROM AFI_APORTES_PRE ap
    INNER JOIN AFI_APORTES_PRE_LOTE apl
        ON apl.prepaga = ap.prepaga
       AND apl.lote = ap.lote
    CROSS JOIN parametros p
    WHERE ap.prepaga = p.prepaga
      AND apl.cuenta = p.cuenta_os
      AND ap.periodo BETWEEN p.fecha_desde AND p.fecha_hasta
      AND ap.baja_fecha IS NULL
      AND apl.baja_fecha IS NULL
      AND ISNULL(ap.reversion, 'N') = 'N'
    GROUP BY
        ap.prepaga,
        ap.cuit,
        ISNULL(ap.cuenta, 0),
        CONVERT(date, ap.periodo)
),
transferencias AS
(
    SELECT
        ar.prepaga,
        cuit = ar.cuit_contrib,
        cuenta = ISNULL(ar.cuenta, 0),
        periodo = CONVERT(date, ar.periodo),
        transferencias =
            SUM(
                CASE
                    WHEN ar.deb_cred = 'C'
                        THEN ISNULL(ar.valor, 0)
                    WHEN ar.deb_cred = 'D'
                     AND ar.concepto_trans = 'ATC'
                        THEN ISNULL(ar.valor, 0)
                    ELSE ISNULL(ar.valor, 0) * -1
                END
            )
    FROM AFI_APORTES_REAL ar
    INNER JOIN AFI_APORTES_REAL_LOTE arl
        ON arl.prepaga = ar.prepaga
       AND arl.lote = ar.lote
    INNER JOIN APORTE_CONCEPTOS ac
        ON ac.concepto_trans = ar.concepto_trans
    CROSS JOIN parametros p
    WHERE ar.prepaga = p.prepaga
      AND arl.cuenta = p.cuenta_os
      AND ar.periodo BETWEEN p.fecha_desde AND p.fecha_hasta
      AND ar.baja_fecha IS NULL
      AND arl.baja_fecha IS NULL
      AND ac.cta_cte = 'S'
      AND ac.agrup = 'REG'
    GROUP BY
        ar.prepaga,
        ar.cuit_contrib,
        ISNULL(ar.cuenta, 0),
        CONVERT(date, ar.periodo)
),
ajustes AS
(
    SELECT DISTINCT
        ar.prepaga,
        cuit = ar.cuit_contrib,
        cuenta = ISNULL(ar.cuenta, 0),
        periodo = CONVERT(date, ar.periodo),
        ajuste = arl.aju_cta
    FROM AFI_APORTES_REAL ar
    INNER JOIN AFI_APORTES_REAL_LOTE arl
        ON arl.prepaga = ar.prepaga
       AND arl.lote = ar.lote
    INNER JOIN AJU_CTA_SUBCTA acs
        ON acs.aju = arl.aju_cta
    CROSS JOIN parametros p
    WHERE ar.prepaga = p.prepaga
      AND arl.cuenta = p.cuenta_os
      AND ar.periodo BETWEEN p.fecha_desde AND p.fecha_hasta
      AND ar.baja_fecha IS NULL
      AND arl.baja_fecha IS NULL
      AND acs.baja_fecha IS NULL
),
deudas AS
(
    SELECT
        d.prepaga,
        d.cuit,
        d.cuenta,
        c.razon,
        d.periodo,
        d.ddjj_bruta,
        transferencias = ISNULL(t.transferencias, 0),
        deuda = d.ddjj_bruta - ISNULL(t.transferencias, 0),
        f.dia_terminacion,
        p.fecha_calculo
    FROM ddjj d
    INNER JOIN CUENTAS c
        ON c.cuenta = d.cuenta
       AND c.cuit = d.cuit
    INNER JOIN FECHA_VTO_PREV_CUIT f
        ON f.dig_ter_cuit =
           CONVERT(int, SUBSTRING(d.cuit, 11, 1))
       AND f.baja_fecha IS NULL
    CROSS JOIN parametros p
    LEFT JOIN transferencias t
        ON t.prepaga = d.prepaga
       AND t.cuit = d.cuit
       AND t.cuenta = d.cuenta
       AND t.periodo = d.periodo
    LEFT JOIN ajustes a
        ON a.prepaga = d.prepaga
       AND a.cuit = d.cuit
       AND a.cuenta = d.cuenta
       AND a.periodo = d.periodo
    WHERE c.baja_fecha IS NULL
      AND a.ajuste IS NULL
      AND d.ddjj_bruta - ISNULL(t.transferencias, 0) > 0
),
vencimientos_originales AS
(
    SELECT
        d.*,
        vencimiento_original =
            DATEFROMPARTS(
                YEAR(DATEADD(month, 1, d.periodo)),
                MONTH(DATEADD(month, 1, d.periodo)),
                d.dia_terminacion
            )
    FROM deudas d
),
vencimientos AS
(
    SELECT
        v.*,
        vencimiento =
            CASE DATEDIFF(day, '19000101', v.vencimiento_original) % 7
                WHEN 5 THEN DATEADD(day, 2, v.vencimiento_original)
                WHEN 6 THEN DATEADD(day, 1, v.vencimiento_original)
                ELSE v.vencimiento_original
            END
    FROM vencimientos_originales v
),
resultados AS
(
    SELECT
        v.*,

                WHEN DATEDIFF(day, v.vencimiento, v.fecha_calculo) < 0

            END,
        cantidad_periodos =
            COUNT(*) OVER
            (
                PARTITION BY v.prepaga, v.cuit, v.cuenta
            )
    FROM vencimientos v
)

    prepaga,

    cantidad_periodos
```
