```sql
SET NOCOUNT ON

SELECT TOP 20
    a.prepaga,
    a.contra,
    titular = LTRIM(a.ape) + ', ' + LTRIM(a.nombre),
    ac.clase_afi,
    c.compro_tipo,
    c.compro_nro,
    c.fecha,
    c.exen,
    c.grava,
    c.iva_total,
    c.iibb,
    c.iibb2,
    c.iibb3,
    c.cuit,
    genera_iibb_comprobante =
        CASE
            WHEN ISNULL(c.iibb, 0) + ISNULL(c.iibb2, 0) + ISNULL(c.iibb3, 0) > 0 THEN 'S'
            ELSE 'N'
        END,
    padron_arba =
        CASE
            WHEN ia.cuit IS NOT NULL THEN 'S'
            ELSE 'N'
        END,
    padron_caba =
        CASE
            WHEN ic.cuit IS NOT NULL THEN 'S'
            ELSE 'N'
        END
FROM afiliados a
JOIN afi_clase ac
    ON ac.prepaga = a.prepaga
   AND ac.contra = a.contra
   AND ac.clase_afi = 'D'
   AND ac.baja_fecha IS NULL
JOIN comprobantes c
    ON c.prepaga = a.prepaga
   AND c.contra = a.contra
   AND c.compro_tipo = 'FC'
   AND c.baja_fecha IS NULL
LEFT JOIN IIBB_PADRON_ARBA ia
    ON ia.cuit = c.cuit
   AND GETDATE() BETWEEN ia.vigen_desde AND ia.vigen_hasta
   AND ia.alicuota > 0
LEFT JOIN IIBB_PADRON_CABA ic
    ON ic.cuit = c.cuit
   AND GETDATE() BETWEEN ic.vigen_desde AND ic.vigen_hasta
   AND ic.alicuota_percep > 0
WHERE a.inte = '00'
  AND a.baja_fecha IS NULL
  AND SUBSTRING(c.compro_nro, 1, 1) = 'B'
  AND c.exen > 0
  AND ISNULL(c.grava, 0) = 0
  AND (
        ISNULL(c.iibb, 0) + ISNULL(c.iibb2, 0) + ISNULL(c.iibb3, 0) > 0
        OR ia.cuit IS NOT NULL
        OR ic.cuit IS NOT NULL
      )
ORDER BY c.fecha DESC, c.alta_fecha DESC
```
