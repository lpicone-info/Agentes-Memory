### [01/09/2026 16:38:27] Agente

Pato, esta query es para `SMC_Prox_Ver` y busca autorizaciones con el mismo `auto` en más de una `sucur`, agrupadas por tipo, asociadas a usuarios WebPOS activos y con saldo disponible:

```sql
USE SMC_Prox_Ver;
GO

WITH consumo AS (
    SELECT
        ar.sucur_rel AS sucur,
        ar.auto_rel  AS auto,
        apir.nomen,
        apir.prestac,
        SUM(ISNULL(apir.canti, 0)) AS canti_consumida
    FROM auto_remotas ar
    INNER JOIN auto_prestac_items_remotas apir
        ON apir.transac = ar.transac
    WHERE ar.sucur_rel IS NOT NULL
      AND ar.auto_rel  IS NOT NULL
      AND ar.esta = 'A'
      AND ar.baja_fecha IS NULL
      AND apir.baja_fecha IS NULL
    GROUP BY ar.sucur_rel, ar.auto_rel, apir.nomen, apir.prestac

    UNION ALL

    SELECT
        eo.sucur,
        eo.auto,
        eoi.nomen,
        eoi.prestac,
        SUM(ISNULL(eoi.canti, 0)) AS canti_consumida
    FROM expe_ordenes eo
    INNER JOIN expe_orden_items eoi
        ON eoi.ofi   = eo.ofi
       AND eoi.expe  = eo.expe
       AND eoi.orden = eo.orden
    WHERE eo.auto IS NOT NULL
      AND eo.baja_fecha IS NULL
      AND eoi.baja_fecha IS NULL
    GROUP BY eo.sucur, eo.auto, eoi.nomen, eoi.prestac
),
saldo AS (
    SELECT
        a.sucur,
        a.auto,
        a.tipo,
        a.efe_prestad,
        SUM(ISNULL(api.canti, 0)) AS canti_autorizada,
        SUM(ISNULL(c.canti_consumida, 0)) AS canti_consumida,
        SUM(ISNULL(api.canti, 0) - ISNULL(c.canti_consumida, 0)) AS saldo_disponible
    FROM autorizaciones a
    INNER JOIN auto_prestac_items api
        ON api.sucur = a.sucur
       AND api.auto  = a.auto
    LEFT JOIN consumo c
        ON c.sucur   = api.sucur
       AND c.auto    = api.auto
       AND c.nomen   = api.nomen
       AND c.prestac = api.prestac
    WHERE a.baja_fecha IS NULL
      AND api.baja_fecha IS NULL
      AND ISNULL(api.canti, 0) - ISNULL(c.canti_consumida, 0) > 0
    GROUP BY a.sucur, a.auto, a.tipo, a.efe_prestad
),
autos_duplicadas AS (
    SELECT
        auto,
        tipo
    FROM saldo
    GROUP BY auto, tipo
    HAVING COUNT(DISTINCT sucur) > 1
)
SELECT
    s.tipo,
    s.auto,
    COUNT(DISTINCT s.sucur) AS cantidad_sucursales,
    STRING_AGG(CONVERT(VARCHAR(10), s.sucur), ', ') AS sucursales,
    SUM(s.canti_autorizada) AS canti_autorizada,
    SUM(s.canti_consumida) AS canti_consumida,
    SUM(s.saldo_disponible) AS saldo_disponible
FROM saldo s
INNER JOIN autos_duplicadas d
    ON d.auto = s.auto
   AND d.tipo = s.tipo
WHERE EXISTS (
    SELECT 1
    FROM zz_usuari u
    WHERE u.prestad = s.efe_prestad
      AND u.baja_fecha IS NULL
)
GROUP BY s.tipo, s.auto
ORDER BY s.tipo, s.auto;
```

Criterio aplicado: autorización vigente, ítems activos, saldo positivo y al menos un usuario activo asociado al prestador efector de la autorización.

        '0' AS cara,
        SUM(ISNULL(eoi.canti, 0)) AS canti_consumida
    FROM expe_ordenes eo
    INNER JOIN expe_orden_items eoi
        ON eoi.ofi  = eo.ofi
       AND eoi.expe = eo.expe
       AND eoi.item = eo.item
    WHERE eo.sucur IS NOT NULL
      AND eo.auto IS NOT NULL
      AND eo.baja_fecha IS NULL
      AND eoi.baja_fecha IS NULL
      AND eoi.ok = 'S'
    GROUP BY eo.sucur, eo.auto, eoi.nomen, eoi.prestac

    UNION ALL

    SELECT
        eo.sucur,
        eo.auto,
        eoi.nomen,
        eoi.prestac,
        ISNULL(eoi.pieza, 0) AS pieza,
        ISNULL(eoi.cara, '0') AS cara,
        SUM(ISNULL(eoi.canti, 0)) AS canti_consumida
    FROM expe_odon eo
    INNER JOIN expe_odon_items eoi
        ON eoi.ofi  = eo.ofi
       AND eoi.expe = eo.expe
       AND eoi.item = eo.item
    WHERE eo.sucur IS NOT NULL
      AND eo.auto IS NOT NULL
      AND eo.baja_fecha IS NULL
      AND eoi.baja_fecha IS NULL
      AND eoi.ok = 'S'
    GROUP BY eo.sucur, eo.auto, eoi.nomen, eoi.prestac,
             ISNULL(eoi.pieza, 0), ISNULL(eoi.cara, '0')
),
consumos_tot AS (
    SELECT
        sucur,
        auto,
        nomen,
        prestac,
        pieza,
        cara,
        SUM(canti_consumida) AS canti_consumida
    FROM consumos
    GROUP BY sucur, auto, nomen, prestac, pieza, cara
),
autos_con_saldo AS (
    SELECT
        a.sucur,
        a.auto,
        a.tipo,
        ISNULL(a.efe_prestad, a.efe_sana) AS prestad_efector,
        SUM(ISNULL(api.canti, 0)) AS canti_autorizada,
        SUM(ISNULL(c.canti_consumida, 0)) AS canti_consumida,
        SUM(ISNULL(api.canti, 0) - ISNULL(c.canti_consumida, 0)) AS saldo_disponible
    FROM autorizaciones a
    INNER JOIN auto_prestac_items api
        ON api.sucur = a.sucur
       AND api.auto = a.auto
    LEFT JOIN consumos_tot c
        ON c.sucur = api.sucur
       AND c.auto = api.auto
       AND c.nomen = api.nomen
       AND c.prestac = api.prestac
       AND c.pieza = ISNULL(api.pieza, 0)
       AND c.cara = ISNULL(api.cara, '0')
    WHERE a.baja_fecha IS NULL
      AND api.baja_fecha IS NULL
      AND EXISTS (
            SELECT 1
            FROM zz_usuari u
            INNER JOIN prestadores p
                ON p.prestad = u.clini
            WHERE u.tipo = 'W'
              AND u.baja_fecha IS NULL
              AND p.baja_fecha IS NULL
              AND u.clini = ISNULL(a.efe_prestad, a.efe_sana)
      )
    GROUP BY a.sucur, a.auto, a.tipo, ISNULL(a.efe_prestad, a.efe_sana)
    HAVING SUM(ISNULL(api.canti, 0) - ISNULL(c.canti_consumida, 0)) > 0
),
duplicadas AS (
    SELECT tipo, auto
    FROM autos_con_saldo
    GROUP BY tipo, auto
    HAVING COUNT(DISTINCT sucur) > 1
)
SELECT
    s.tipo,
    s.auto,
    COUNT(DISTINCT s.sucur) AS cantidad_sucursales,
    COUNT(DISTINCT s.prestad_efector) AS cantidad_prestadores,
    SUM(s.canti_autorizada) AS canti_autorizada,
    SUM(s.canti_consumida) AS canti_consumida,
    SUM(s.saldo_disponible) AS saldo_disponible
FROM autos_con_saldo s
INNER JOIN duplicadas d
    ON d.tipo = s.tipo
   AND d.auto = s.auto
GROUP BY s.tipo, s.auto
ORDER BY s.tipo, s.auto;
