  IF OBJECT_ID('tempdb..#TMP_TRIGGERS') IS NOT NULL DROP TABLE #TMP_TRIGGERS;
  SELECT  DISTINCT    -- Triggers habilitados
          tabla                           = dat.tabla,
          trigger_name                    = dat.trigger_name,
          codigo                          = dat.codigo
  INTO    #TMP_TRIGGERS
  FROM    SYS.TRIGGERS                    as tr
        
          INNER JOIN 
          SYS.SQL_EXPRESSION_DEPENDENCIES as dep
          ON dep.referencing_id           = tr.object_id
          INNER JOIN
          (
            SELECT  DISTINCT
                    tabla,
                    object_id     = OBJECT_ID(tabla)
            FROM    #TMP_OFUSCACION
          )                       as tbl
          ON dep.referenced_id            = tbl.object_id

          CROSS APPLY
          (
            SELECT
                    tabla           = OBJECT_NAME(tr.parent_id),
                    trigger_name    = tr.name,
                    codigo          = OBJECT_DEFINITION(tr.object_id)
          )                               as dat

  WHERE   tr.is_disabled                  = 0
          AND EXISTS(
                      SELECT  1
                      FROM    #TMP_OFUSCACION
                      WHERE   tabla       = dat.tabla
                    )
  UNION
  SELECT  DISTINCT
          tabla                           = OBJECT_NAME(tr.parent_id),
          trigger_name                    = tr.name,
          codigo                          = OBJECT_DEFINITION(tr.object_id)

  FROM    SYS.TRIGGERS                    as tr
          INNER JOIN
          (
            SELECT  DISTINCT
                    tabla,
                    object_id     = OBJECT_ID(tabla)
            FROM    #TMP_OFUSCACION
          )                       as tbl
          ON tr.parent_id         = tbl.object_id

          CROSS APPLY
          (
            SELECT
                    tabla           = OBJECT_NAME(tr.parent_id),
                    trigger_name    = tr.name,
                    codigo          = OBJECT_DEFINITION(tr.object_id)
          )                               as dat

  WHERE   tr.is_disabled          = 0
          AND EXISTS(
                      SELECT  1
                      FROM    #TMP_OFUSCACION
                      WHERE   tabla       = dat.tabla
                    );

  DECLARE cur_ofuscacion CURSOR LOCAL FAST_FORWARD FOR

                                                                                (

                                                                                  FROM    #TMP_OFUSCACION     as x

                                                    FROM  (
                                                            SELECT  DISTINCT

                                                                    tabla
                                                            FROM    #TMP_OFUSCACION
                                                          ) as y
                                                    ORDER 
                                                    BY      y.esquema, y.tabla;
