  DECLARE

          @ScriptN            nvarchar(max),

          @ArchivoLog         sysname,

  IF @Ejecutar = 1 AND @ReducirLog = 1
  BEGIN
    SELECT  @RecoveryAnterior = recovery_model_desc
    FROM    sys.databases
    WHERE   database_id = DB_ID();

    SELECT  TOP 1

    FROM    sys.database_files
    WHERE   type_desc = 'LOG'

  DECLARE cur_ofuscacion CURSOR LOCAL FAST_FORWARD FOR
                                                    SELECT  tabla       = QUOTENAME(y.esquema) + '.' + QUOTENAME(y.tabla),
                                                            script      = STUFF(
                                                                                (
                                                                                  SELECT  ', ' + x.asignacion
                                                                                  FROM    #TMP_OFUSCACION     as x
                                                                                  WHERE   x.tabla = y.tabla   FOR XML PATH(''), TYPE
                                                                                ).value('.', 'varchar(max)'), 1, 2, ''
                                                                               ),
                                                            condition   = STUFF(
                                                                                (
                                                                                  SELECT  ' OR ' + x.condicion
                                                                                  FROM    #TMP_OFUSCACION     as x
                                                                                  WHERE   x.tabla = y.tabla   FOR XML PATH(''), TYPE
                                                                                ).value('.', 'varchar(max)'), 2, 2, ''
                                                                               )
                                                    FROM  (
                                                            SELECT  DISTINCT
                                                                    esquema, 
                                                                    tabla
                                                            FROM    #TMP_OFUSCACION
                                                          ) as y
                                                    ORDER 
                                                    BY      y.esquema, y.tabla;

  OPEN cur_ofuscacion;
  FETCH NEXT FROM cur_ofuscacion INTO @Tabla, @Script, @Condicion;

  SET @TotalFilas = 0;

    SET @Condicion = '1 = 1';
    SET @ScriptEjecutable = 'SET ROWCOUNT 0;' +

                            'SET    ' + @Script + ' ' +
                            'FROM   ' + @Tabla + ' as tgt ' +
                            'WHERE  ' + @Condicion + '; ' +

                            'SET ROWCOUNT 0';

    BEGIN

        SET @ScriptN = CONVERT(nvarchar(max), @ScriptEjecutable);
        EXEC sys.sp_executesql @ScriptN, N'@FilasActual integer OUTPUT', @FilasActual = @Filas OUTPUT;

        SET @TotalFilas = @TotalFilas + @Filas
        IF @ReducirLog = 1 AND @Filas > 0
        BEGIN
          SET @Msg = 'Haciendo CheckPoint';
          RAISERROR(@Msg, 0, 0) WITH NOWAIT;
          CHECKPOINT;
        END

    END

    FETCH NEXT FROM cur_ofuscacion INTO @Tabla, @Script, @Condicion;

  END

  CLOSE cur_ofuscacion;
  DEALLOCATE cur_ofuscacion;
