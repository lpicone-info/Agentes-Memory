INSERT INTO #MAXIMO_FECHA_LETRA_PUNTO
(
    ente,
    compro_tipo,
    compro_letra,
    compro_punto,
    max_fecha
)
SELECT  DISTINCT
        ente          = COM.ente,
        compro_tipo   = COM.compro_tipo,
        compro_letra  = SUBSTRING(COM.compro_nro,1,1),
        compro_punto  = SUBSTRING(COM.compro_nro,2,4),
        max_fecha     = MAX(COM.fecha)
FROM    #DISTINCT_LETRA_PUNTO AS CAG
        INNER JOIN COMPROBANTES AS COM
        ON  COM.ente        = CAG.ente
        AND COM.compro_tipo = CAG.compro_tipo
        AND COM.compro_nro LIKE CAG.compro_letra_punto + '%'
WHERE   COM.fecha > @prm_fecha
GROUP BY COM.ente,
         COM.compro_tipo,
         SUBSTRING(COM.compro_nro,1,1),
         SUBSTRING(COM.compro_nro,2,4)

UPDATE  #FAC_COMPRO_A_GENERAR
SET     fecha      = MAP.max_fecha,
        vto1_fecha = MAP.max_fecha
FROM    #FAC_COMPRO_A_GENERAR AS CAG
        INNER JOIN #MAXIMO_FECHA_LETRA_PUNTO AS MAP
        ON  MAP.ente         = CAG.ente
        AND MAP.compro_tipo  = CAG.compro_tipo
        AND MAP.compro_letra = CAG.compro_letra
