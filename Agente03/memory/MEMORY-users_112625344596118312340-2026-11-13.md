public function integer of_alicuota_iva (integer ai_prepaga, string as_contra, datetime adt_fecha, ref decimal adc_alicuota);LONG			ll_iva
STRING		ls_es_inscrip , ls_es_grava,ls_es_gravaN, ls_msg
DECIMAL		ldc_iva_tasa, ldc_ivan_tasa

adc_alicuota = 0

// Verifico si la prepaga discrimina IVA.
SELECT  	es_inscrip   
INTO		:ls_es_inscrip 
FROM    	prepagas 	p, 
			entes		e   
WHERE   	prepaga 	= :ai_prepaga   
AND     	p.ente  	= e.ente;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar la condición ante el IVA de la prepaga."	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Si no discrimina Iva no hay nada mas que hacer.
IF ls_es_inscrip  <> 'S' THEN RETURN 0

// Verifico si el Contrato es gravado ante el Iva.
SELECT 	i.es_grava   
INTO		:ls_es_grava
FROM   	afi_gravados 	a,   
			afiliados 		f,
			ivas  			i
WHERE  	a.prepaga 		= :ai_prepaga   
AND    	a.contra  		= :as_contra   
AND    	a.iva   			= i.iva   
AND    	a.prepaga 		= f.prepaga 
AND    	a.contra  		= f.contra
AND		a.inte    		= f.inte
AND    	f.paren 			= 'T'
AND    	vigen_desde 	= ( SELECT 	max( w.vigen_desde )   
									 FROM  	afi_gravados 	w 
									 WHERE 	w.prepaga 		= a.prepaga   
									 AND   	w.contra  		= a.contra   
									 AND   	w.inte    		= a.inte
									 AND   	w.vigen_Desde 	<= :adt_fecha
									 AND   	(w.baja_fecha 	IS null 
										OR 	w.baja_fecha > :adt_fecha )
									) ;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condicion ante el IVA del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Verifico si aplica Iva No Inscripto.
SELECT   i.es_grava   
INTO		:ls_es_gravaN
FROM  	parametros		p,
			ivas				i
WHERE		p.ivan_inscrip = 	i.iva 
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condición ante el IVA n del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

SELECT  	iva_tasa, 
			ivan_tasa   
INTO		:ldc_iva_tasa, 	
			:ldc_ivan_tasa
FROM    	iva_tasa   
WHERE   	vigen_desde = (SELECT 	max(vigen_desde) 
								FROM 		iva_tasa   
								WHERE  	vigen_desde 	<= :adt_fecha 
								AND   	(baja_fecha 	IS null 
										OR 	baja_fecha 	> :adt_fecha ))
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al traer las alícuotas de IVA"	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

IF ls_es_grava = 'S' THEN
	adc_alicuota = ldc_iva_tasa / 100
END IF

IF ls_es_grava = 'S' AND ls_es_gravaN = 'S' THEN
	adc_alicuota = adc_alicuota + (ldc_ivan_tasa / 100)
END IF

RETURN 0
end function

public function boolean of_debe_generar_compro (integer ai_prepaga, string as_contra, integer ai_sucur, long al_auto, string as_factu_concep_cop);LONG ll_cuenta
DATETIME ldt_fecha
STRING ls_modo_fac
LONG ll_aju_cop

setNull(ll_cuenta)
setNull(ll_aju_cop)
setNull(ls_modo_fac)

SELECT fecha
INTO :ldt_fecha
FROM AUTORIZACIONES
WHERE 	sucur = :ai_sucur
AND		auto = :al_auto;

SELECT AC.cuenta, SC.modo_fac
INTO	:ll_cuenta, :ls_modo_fac
FROM AFI_CLASE	AC

		INNER JOIN
		subcta_contratos	SC
		ON	 	AC.cuenta 	= SC.cuenta
		AND	  AC.subcta	= SC.subcta
		
WHERE 	AC	.prepaga =  :ai_prepaga
AND		AC.contra	=  :as_contra
AND		AC.baja_fecha IS NULL
AND		SC.baja_fecha IS NULL
AND		AC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM AFI_CLASE 
										WHERE 	prepaga = AC.prepaga
										AND		contra	= AC.contra
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                      	)
AND		SC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM subcta_contratos 
										WHERE 	cuenta 	= AC.cuenta
										AND		subcta	= AC.subcta
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                        )
                        									
AND		AC.clase_afi = 'E'
AND		(	SC.modo_fac = 'E'
					OR
				(
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'E' AND
				    NOT EXISTS(SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )
				  OR
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'A' AND
				    EXISTS    (SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )            
				)
			);

IF SQLCA.sqlcode < 0 THEN
	MessageBox('Error', 'Error al consultar los datos de la cuenta - subcuenta~n' + SQLCA.sqlerrtext)
	RETURN FALSE
END IF

IF NOT(isNull(ll_cuenta))	THEN
	//NO DEBE GENERAR COMPROBANTE: FACTURACION A EMPRESA O FACTURACION MIXTA CON COPAGO A LA EMPRESA

	IF ls_modo_fac = 'E' THEN RETURN TRUE //R29684
	RETURN FALSE
ELSE	
	RETURN TRUE
END IF	
end function

public function boolean of_compro_cae (integer li_ente, string ls_tipo, string ls_nro);BOOLEAN 			ib_emite_facturas_electronicas, ib_permite_imprimir_sin_CAE ,ib_hay_facturas_sin_cae,lb_exist			     														
STRING 				ls_falta_cae,ls_msg 
INTEGER      			ll_n

str_compro astr[]
									            			    							
uo_nv_afip_wsfe_autrequest lou_autrequest

ib_emite_facturas_electronicas = FALSE
ib_permite_imprimir_sin_CAE = FALSE

IF upperbound(glob.gs_reglas) >= 198 THEN
	IF glob.gs_reglas[198] = 'S' THEN ib_emite_facturas_electronicas = TRUE
END IF

IF upperbound(glob.gs_reglas) >= 199 THEN
	IF glob.gs_reglas[199] = 'S' THEN ib_permite_imprimir_sin_CAE = TRUE
END IF

IF ib_emite_facturas_electronicas THEN 									
	ib_hay_facturas_sin_cae = FALSE 										
	
	DECLARE falta_cae PROCEDURE FOR dbo.afs_compro_falta_cae
		@ente			 = :li_ente,
		@compro_tipo = :ls_tipo,    
		@compro_nro  = :ls_nro,    			
		@falta_cae 	 = :ls_falta_cae OUTPUT;
			
		astr[1].ente 				= li_ente
		astr[1].compro_tipo  	= ls_tipo 	
		astr[1].compro_nro  	= ls_nro 	
	
		
				
		EXECUTE falta_cae;
		
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (EXECUTE)',2,0)
			ROLLBACK USING sqlca;
			RETURN FALSE
		END IF
		
		FETCH falta_cae
		INTO :ls_falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (FETCH)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		CLOSE falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (CLOSE)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		IF ls_falta_cae = 'S' THEN	ib_hay_facturas_sin_cae = TRUE
		
	END IF
	
	COMMIT;		//Esto es para no interferir con el manejo de trasacciones que hace el Wwb Service.
	
	IF ib_hay_facturas_sin_cae THEN
		IF glob.of_valor_regla(220) = 'W' THEN 
			//Solicitamos a la AFIP el CAE del comprobante utilizando el WebService.			
			lou_autrequest = CREATE uo_nv_afip_wsfe_autrequest
			
			IF lou_autrequest.EVENT ue_inicializar() THEN
				
				IF lou_autrequest.EVENT ue_solicitar_cae(astr) THEN
					ib_hay_facturas_sin_cae = FALSE	//Obtuvimos todos los CAEs faltantes.
				ELSE
					mbox(gw_ventana, 'Ocurrio un error al solicitar el CAE a la AFIP utilizando el Web Service:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)
				END IF
				
			ELSE				
				mbox(gw_ventana, 'Ocurrio un error al inicializar el Web Service de Facturacion Electronica:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)				
			END IF							
		END IF
		
		//Si siguen existiendo comprobantes sin CAE...
		IF ib_hay_facturas_sin_cae THEN
			ls_msg = 'El comprobante ' + ls_tipo + ' ' + ls_nro + ' necesita el Codigo de Autorización Electronico (CAE) emitido por la AFIP.'
			IF ib_permite_imprimir_sin_CAE THEN
				IF mbox(gw_ventana, ls_msg + '~n~rSin el CAE este comprobante carece de valor fiscal.~n~r¿Desea imprimirlo de todas formas?',4,2) = 2 THEN
					RETURN FALSE
				END IF
			ELSE
				mbox(gw_ventana, ls_msg,1,0)
				RETURN 	FALSE
			END IF
		END IF
	END IF	

COMMIT;		

RETURN TRUE

public function integer of_alicuota_iva (integer ai_prepaga, string as_contra, datetime adt_fecha, ref decimal adc_alicuota);LONG			ll_iva
STRING		ls_es_inscrip , ls_es_grava,ls_es_gravaN, ls_msg
DECIMAL		ldc_iva_tasa, ldc_ivan_tasa

adc_alicuota = 0

// Verifico si la prepaga discrimina IVA.
SELECT  	es_inscrip   
INTO		:ls_es_inscrip 
FROM    	prepagas 	p, 
			entes		e   
WHERE   	prepaga 	= :ai_prepaga   
AND     	p.ente  	= e.ente;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar la condición ante el IVA de la prepaga."	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Si no discrimina Iva no hay nada mas que hacer.
IF ls_es_inscrip  <> 'S' THEN RETURN 0

// Verifico si el Contrato es gravado ante el Iva.
SELECT 	i.es_grava   
INTO		:ls_es_grava
FROM   	afi_gravados 	a,   
			afiliados 		f,
			ivas  			i
WHERE  	a.prepaga 		= :ai_prepaga   
AND    	a.contra  		= :as_contra   
AND    	a.iva   			= i.iva   
AND    	a.prepaga 		= f.prepaga 
AND    	a.contra  		= f.contra
AND		a.inte    		= f.inte
AND    	f.paren 			= 'T'
AND    	vigen_desde 	= ( SELECT 	max( w.vigen_desde )   
									 FROM  	afi_gravados 	w 
									 WHERE 	w.prepaga 		= a.prepaga   
									 AND   	w.contra  		= a.contra   
									 AND   	w.inte    		= a.inte
									 AND   	w.vigen_Desde 	<= :adt_fecha
									 AND   	(w.baja_fecha 	IS null 
										OR 	w.baja_fecha > :adt_fecha )
									) ;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condicion ante el IVA del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Verifico si aplica Iva No Inscripto.
SELECT   i.es_grava   
INTO		:ls_es_gravaN
FROM  	parametros		p,
			ivas				i
WHERE		p.ivan_inscrip = 	i.iva 
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condición ante el IVA n del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

SELECT  	iva_tasa, 
			ivan_tasa   
INTO		:ldc_iva_tasa, 	
			:ldc_ivan_tasa
FROM    	iva_tasa   
WHERE   	vigen_desde = (SELECT 	max(vigen_desde) 
								FROM 		iva_tasa   
								WHERE  	vigen_desde 	<= :adt_fecha 
								AND   	(baja_fecha 	IS null 
										OR 	baja_fecha 	> :adt_fecha ))
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al traer las alícuotas de IVA"	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

IF ls_es_grava = 'S' THEN
	adc_alicuota = ldc_iva_tasa / 100
END IF

IF ls_es_grava = 'S' AND ls_es_gravaN = 'S' THEN
	adc_alicuota = adc_alicuota + (ldc_ivan_tasa / 100)
END IF

RETURN 0
end function

DATETIME ldt_fecha
STRING ls_modo_fac

setNull(ll_cuenta)
setNull(ls_modo_fac)

SELECT fecha
INTO :ldt_fecha
FROM AUTORIZACIONES
WHERE 	sucur = :ai_sucur
AND		auto = :al_auto;

SELECT AC.cuenta, SC.modo_fac
INTO	:ll_cuenta, :ls_modo_fac
FROM AFI_CLASE	AC

		INNER JOIN
		subcta_contratos	SC
		ON	 	AC.cuenta 	= SC.cuenta
		AND	  AC.subcta	= SC.subcta
		
WHERE 	AC	.prepaga =  :ai_prepaga
AND		AC.contra	=  :as_contra
AND		AC.baja_fecha IS NULL
AND		SC.baja_fecha IS NULL
AND		AC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM AFI_CLASE 
										WHERE 	prepaga = AC.prepaga
										AND		contra	= AC.contra
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                      	)
AND		SC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM subcta_contratos 
										WHERE 	cuenta 	= AC.cuenta
										AND		subcta	= AC.subcta
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                        )
                        									
AND		AC.clase_afi = 'E'
AND		(	SC.modo_fac = 'E'
					OR
				(
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'E' AND
				    NOT EXISTS(SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )
				  OR
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'A' AND
				    EXISTS    (SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )            
				)
			);

IF SQLCA.sqlcode < 0 THEN
	MessageBox('Error', 'Error al consultar los datos de la cuenta - subcuenta~n' + SQLCA.sqlerrtext)
	RETURN FALSE
END IF

IF NOT(isNull(ll_cuenta))	THEN
	//NO DEBE GENERAR COMPROBANTE: FACTURACION A EMPRESA O FACTURACION MIXTA CON COPAGO A LA EMPRESA

	IF al_auto > 0 AND ls_modo_fac = 'E' THEN RETURN TRUE //R29684
	RETURN FALSE
ELSE	
	RETURN TRUE
END IF	
end function

public function boolean of_compro_cae (integer li_ente, string ls_tipo, string ls_nro);BOOLEAN 			ib_emite_facturas_electronicas, ib_permite_imprimir_sin_CAE ,ib_hay_facturas_sin_cae,lb_exist			     														
STRING 				ls_falta_cae,ls_msg 
INTEGER      			ll_n

str_compro astr[]
									            			    							
uo_nv_afip_wsfe_autrequest lou_autrequest

ib_emite_facturas_electronicas = FALSE
ib_permite_imprimir_sin_CAE = FALSE

IF upperbound(glob.gs_reglas) >= 198 THEN
	IF glob.gs_reglas[198] = 'S' THEN ib_emite_facturas_electronicas = TRUE
END IF

IF upperbound(glob.gs_reglas) >= 199 THEN
	IF glob.gs_reglas[199] = 'S' THEN ib_permite_imprimir_sin_CAE = TRUE
END IF

IF ib_emite_facturas_electronicas THEN 									
	ib_hay_facturas_sin_cae = FALSE 										
	
	DECLARE falta_cae PROCEDURE FOR dbo.afs_compro_falta_cae
		@ente			 = :li_ente,
		@compro_tipo = :ls_tipo,    
		@compro_nro  = :ls_nro,    			
		@falta_cae 	 = :ls_falta_cae OUTPUT;
			
		astr[1].ente 				= li_ente
		astr[1].compro_tipo  	= ls_tipo 	
		astr[1].compro_nro  	= ls_nro 	
	
		
				
		EXECUTE falta_cae;
		
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (EXECUTE)',2,0)
			ROLLBACK USING sqlca;
			RETURN FALSE
		END IF
		
		FETCH falta_cae
		INTO :ls_falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (FETCH)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		CLOSE falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (CLOSE)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		IF ls_falta_cae = 'S' THEN	ib_hay_facturas_sin_cae = TRUE
		
	END IF
	
	COMMIT;		//Esto es para no interferir con el manejo de trasacciones que hace el Wwb Service.
	
	IF ib_hay_facturas_sin_cae THEN
		IF glob.of_valor_regla(220) = 'W' THEN 
			//Solicitamos a la AFIP el CAE del comprobante utilizando el WebService.			
			lou_autrequest = CREATE uo_nv_afip_wsfe_autrequest
			
			IF lou_autrequest.EVENT ue_inicializar() THEN
				
				IF lou_autrequest.EVENT ue_solicitar_cae(astr) THEN
					ib_hay_facturas_sin_cae = FALSE	//Obtuvimos todos los CAEs faltantes.
				ELSE
					mbox(gw_ventana, 'Ocurrio un error al solicitar el CAE a la AFIP utilizando el Web Service:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)
				END IF
				
			ELSE				
				mbox(gw_ventana, 'Ocurrio un error al inicializar el Web Service de Facturacion Electronica:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)				
			END IF							
		END IF
		
		//Si siguen existiendo comprobantes sin CAE...
		IF ib_hay_facturas_sin_cae THEN
			ls_msg = 'El comprobante ' + ls_tipo + ' ' + ls_nro + ' necesita el Codigo de Autorización Electronico (CAE) emitido por la AFIP.'
			IF ib_permite_imprimir_sin_CAE THEN
				IF mbox(gw_ventana, ls_msg + '~n~rSin el CAE este comprobante carece de valor fiscal.~n~r¿Desea imprimirlo de todas formas?',4,2) = 2 THEN
					RETURN FALSE
				END IF
			ELSE
				mbox(gw_ventana, ls_msg,1,0)
				RETURN 	FALSE
			END IF
		END IF
	END IF	

COMMIT;		

RETURN TRUE
end function

public function integer of_alicuota_iva (integer ai_prepaga, string as_contra, datetime adt_fecha, ref decimal adc_alicuota);LONG			ll_iva
STRING		ls_es_inscrip , ls_es_grava,ls_es_gravaN, ls_msg
DECIMAL		ldc_iva_tasa, ldc_ivan_tasa

adc_alicuota = 0

// Verifico si la prepaga discrimina IVA.
SELECT  	es_inscrip   
INTO		:ls_es_inscrip 
FROM    	prepagas 	p, 
			entes		e   
WHERE   	prepaga 	= :ai_prepaga   
AND     	p.ente  	= e.ente;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar la condición ante el IVA de la prepaga."	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Si no discrimina Iva no hay nada mas que hacer.
IF ls_es_inscrip  <> 'S' THEN RETURN 0

// Verifico si el Contrato es gravado ante el Iva.
SELECT 	i.es_grava   
INTO		:ls_es_grava
FROM   	afi_gravados 	a,   
			afiliados 		f,
			ivas  			i
WHERE  	a.prepaga 		= :ai_prepaga   
AND    	a.contra  		= :as_contra   
AND    	a.iva   			= i.iva   
AND    	a.prepaga 		= f.prepaga 
AND    	a.contra  		= f.contra
AND		a.inte    		= f.inte
AND    	f.paren 			= 'T'
AND    	vigen_desde 	= ( SELECT 	max( w.vigen_desde )   
									 FROM  	afi_gravados 	w 
									 WHERE 	w.prepaga 		= a.prepaga   
									 AND   	w.contra  		= a.contra   
									 AND   	w.inte    		= a.inte
									 AND   	w.vigen_Desde 	<= :adt_fecha
									 AND   	(w.baja_fecha 	IS null 
										OR 	w.baja_fecha > :adt_fecha )
									) ;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condicion ante el IVA del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

// Verifico si aplica Iva No Inscripto.
SELECT   i.es_grava   
INTO		:ls_es_gravaN
FROM  	parametros		p,
			ivas				i
WHERE		p.ivan_inscrip = 	i.iva 
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al Verificar Condición ante el IVA n del contrato "	 + as_contra + "."
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

SELECT  	iva_tasa, 
			ivan_tasa   
INTO		:ldc_iva_tasa, 	
			:ldc_ivan_tasa
FROM    	iva_tasa   
WHERE   	vigen_desde = (SELECT 	max(vigen_desde) 
								FROM 		iva_tasa   
								WHERE  	vigen_desde 	<= :adt_fecha 
								AND   	(baja_fecha 	IS null 
										OR 	baja_fecha 	> :adt_fecha ))
;

IF SQLCA.SqlCode < 0 THEN
	ls_msg = "Error al traer las alícuotas de IVA"	
	Mbox(gw_ventana, ls_msg, 2, 0 )	
	RETURN -1
END IF

IF ls_es_grava = 'S' THEN
	adc_alicuota = ldc_iva_tasa / 100
END IF

IF ls_es_grava = 'S' AND ls_es_gravaN = 'S' THEN
	adc_alicuota = adc_alicuota + (ldc_ivan_tasa / 100)
END IF

RETURN 0
end function

public function boolean of_debe_generar_compro (integer ai_prepaga, string as_contra, integer ai_sucur, long al_auto, string as_factu_concep_cop);LONG ll_cuenta
DATETIME ldt_fecha
STRING ls_modo_fac

setNull(ll_cuenta)
setNull(ls_modo_fac)

SELECT fecha
INTO :ldt_fecha
FROM AUTORIZACIONES
WHERE 	sucur = :ai_sucur
AND		auto = :al_auto;

SELECT AC.cuenta, SC.modo_fac
INTO	:ll_cuenta, :ls_modo_fac
FROM AFI_CLASE	AC

		INNER JOIN
		subcta_contratos	SC
		ON	 	AC.cuenta 	= SC.cuenta
		AND	  AC.subcta	= SC.subcta
		
WHERE 	AC	.prepaga =  :ai_prepaga
AND		AC.contra	=  :as_contra
AND		AC.baja_fecha IS NULL
AND		SC.baja_fecha IS NULL
AND		AC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM AFI_CLASE 
										WHERE 	prepaga = AC.prepaga
										AND		contra	= AC.contra
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                      	)
AND		SC.vigen_desde = (
										SELECT MAX(vigen_desde)
										FROM subcta_contratos 
										WHERE 	cuenta 	= AC.cuenta
										AND		subcta	= AC.subcta
										AND		vigen_desde <= :ldt_fecha
										AND		baja_fecha IS NULL
                        )
                        									
AND		AC.clase_afi = 'E'
AND		(	SC.modo_fac = 'E'
					OR
				(
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'E' AND
				    NOT EXISTS(SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )
				  OR
				  (
					  SC.modo_fac = 'M'     AND
					  SC.modo_fac_def = 'A' AND
				    EXISTS    (SELECT 1
				               FROM subcta_conceptos
				               WHERE cuenta       = SC.cuenta
				               AND   subcta       = SC.subcta
				               AND   vigen_desde  = SC.vigen_desde
				               AND   baja_fecha IS NULL
				               AND   concep = :as_factu_concep_cop
				               )
				  )            
				)
			);

IF SQLCA.sqlcode < 0 THEN
	MessageBox('Error', 'Error al consultar los datos de la cuenta - subcuenta~n' + SQLCA.sqlerrtext)
	RETURN FALSE
END IF

IF NOT(isNull(ll_cuenta))	THEN
	IF al_auto > 0 AND as_factu_concep_cop = 'COP' AND ls_modo_fac = 'E' THEN RETURN TRUE //R29684: Esta condicion ahora SI debe generar el comprobante.
	
	//NO DEBE GENERAR COMPROBANTE: FACTURACION A EMPRESA O FACTURACION MIXTA CON COPAGO A LA EMPRESA
	RETURN FALSE
ELSE	
	RETURN TRUE
END IF	
end function

public function boolean of_compro_cae (integer li_ente, string ls_tipo, string ls_nro);BOOLEAN 			ib_emite_facturas_electronicas, ib_permite_imprimir_sin_CAE ,ib_hay_facturas_sin_cae,lb_exist			     														
STRING 				ls_falta_cae,ls_msg 
INTEGER      			ll_n

str_compro astr[]
									            			    							
uo_nv_afip_wsfe_autrequest lou_autrequest

ib_emite_facturas_electronicas = FALSE
ib_permite_imprimir_sin_CAE = FALSE

IF upperbound(glob.gs_reglas) >= 198 THEN
	IF glob.gs_reglas[198] = 'S' THEN ib_emite_facturas_electronicas = TRUE
END IF

IF upperbound(glob.gs_reglas) >= 199 THEN
	IF glob.gs_reglas[199] = 'S' THEN ib_permite_imprimir_sin_CAE = TRUE
END IF

IF ib_emite_facturas_electronicas THEN 									
	ib_hay_facturas_sin_cae = FALSE 										
	
	DECLARE falta_cae PROCEDURE FOR dbo.afs_compro_falta_cae
		@ente			 = :li_ente,
		@compro_tipo = :ls_tipo,    
		@compro_nro  = :ls_nro,    			
		@falta_cae 	 = :ls_falta_cae OUTPUT;
			
		astr[1].ente 				= li_ente
		astr[1].compro_tipo  	= ls_tipo 	
		astr[1].compro_nro  	= ls_nro 	
	
		
				
		EXECUTE falta_cae;
		
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (EXECUTE)',2,0)
			ROLLBACK USING sqlca;
			RETURN FALSE
		END IF
		
		FETCH falta_cae
		INTO :ls_falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (FETCH)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		CLOSE falta_cae;
		IF sqlca.sqlcode < 0 THEN
			mbox(gw_ventana, 'Ocurrio un error al averiguar si al comprobante le falta el CAE. (CLOSE)',2,0)
			ROLLBACK USING sqlca;		
			RETURN FALSE
		END IF
		
		IF ls_falta_cae = 'S' THEN	ib_hay_facturas_sin_cae = TRUE
		
	END IF
	
	COMMIT;		//Esto es para no interferir con el manejo de trasacciones que hace el Wwb Service.
	
	IF ib_hay_facturas_sin_cae THEN
		IF glob.of_valor_regla(220) = 'W' THEN 
			//Solicitamos a la AFIP el CAE del comprobante utilizando el WebService.			
			lou_autrequest = CREATE uo_nv_afip_wsfe_autrequest
			
			IF lou_autrequest.EVENT ue_inicializar() THEN
				
				IF lou_autrequest.EVENT ue_solicitar_cae(astr) THEN
					ib_hay_facturas_sin_cae = FALSE	//Obtuvimos todos los CAEs faltantes.
				ELSE
					mbox(gw_ventana, 'Ocurrio un error al solicitar el CAE a la AFIP utilizando el Web Service:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)
				END IF
				
			ELSE				
				mbox(gw_ventana, 'Ocurrio un error al inicializar el Web Service de Facturacion Electronica:~n~r' + lou_autrequest.EVENT ue_errores_primeros(8),1,0)				
			END IF							
		END IF
		
		//Si siguen existiendo comprobantes sin CAE...
		IF ib_hay_facturas_sin_cae THEN
			ls_msg = 'El comprobante ' + ls_tipo + ' ' + ls_nro + ' necesita el Codigo de Autorización Electronico (CAE) emitido por la AFIP.'
			IF ib_permite_imprimir_sin_CAE THEN
				IF mbox(gw_ventana, ls_msg + '~n~rSin el CAE este comprobante carece de valor fiscal.~n~r¿Desea imprimirlo de todas formas?',4,2) = 2 THEN
					RETURN FALSE
				END IF
			ELSE
				mbox(gw_ventana, ls_msg,1,0)
				RETURN 	FALSE
			END IF
		END IF
	END IF	

COMMIT;		

RETURN TRUE
end function
