```sql
SELECT x.cuit,
       x.cuil,
       x.periodo,
       x.lote_mayor_alta,
       x.registro_mayor_alta,
       x.mayor_alta_fecha,
       x.origrect_en_mayor_alta,
       x.lote_mayor_origrect,
       x.registro_mayor_origrect,
       x.alta_fecha_en_mayor_origrect,
       x.mayor_origrect
FROM (
    SELECT m.cuit,
           m.cuil,
           m.periodo,
           a.lote AS lote_mayor_alta,
           a.registro AS registro_mayor_alta,
           a.alta_fecha AS mayor_alta_fecha,
           a.origrect AS origrect_en_mayor_alta,
           o.lote AS lote_mayor_origrect,
           o.registro AS registro_mayor_origrect,
           o.alta_fecha AS alta_fecha_en_mayor_origrect,
           o.origrect AS mayor_origrect
    FROM (
        SELECT cuit,
               cuil,
               periodo,
               MAX(alta_fecha) AS max_alta_fecha,
               MAX(origrect) AS max_origrect
        FROM dbo.afi_aportes_pre
        WHERE baja_fecha IS NULL
        GROUP BY cuit,
                 cuil,
                 periodo
        HAVING COUNT(*) > 1
    ) m
    JOIN dbo.afi_aportes_pre a
      ON a.cuit = m.cuit
     AND a.cuil = m.cuil
     AND a.periodo = m.periodo
     AND a.alta_fecha = m.max_alta_fecha
     AND a.baja_fecha IS NULL
    JOIN dbo.afi_aportes_pre o
      ON o.cuit = m.cuit
     AND o.cuil = m.cuil
     AND o.periodo = m.periodo
     AND o.origrect = m.max_origrect
     AND o.baja_fecha IS NULL
    WHERE (a.lote <> o.lote OR a.registro <> o.registro)
      AND a.lote <> o.lote
      AND ISNULL(a.origrect, -1) <> ISNULL(o.origrect, -1)
) x
ORDER BY x.cuit,
         x.cuil,
         x.periodo
```
