```sql
CREATE PROCEDURE dbo.prs_expe_farma_lotes
    @prm_ofi smallint,
    @prm_expe integer,
    @prm_farma integer = null
AS
BEGIN

    SET COMPATIBILITY_MODE OFF

    SELECT  ef.farma,
            f.manda_farma,
            f.razon,
            ef.lote,
            ef.deno,
            ef.canti,
            ef.total,
            ef.presen_fecha,
            ef.rece_nro,
            ef.ok,
            ef.alta_fecha,
            ef.carga_fecha,
            ef.actu_fecha,
            ef.afi_dto,
            ef.nro_caja
    FROM    expe_farmacias ef
            INNER JOIN farmacias f
                ON f.farma = ef.farma
               AND f.baja_fecha IS NULL
    WHERE   ef.ofi = @prm_ofi
    AND     ef.expe = @prm_expe
    AND     ef.baja_fecha IS NULL
    AND     ( @prm_farma IS NULL OR @prm_farma = 0 OR ef.farma = @prm_farma )

END
```
